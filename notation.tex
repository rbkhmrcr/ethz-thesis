\chapter*{Notation}\addcontentsline{toc}{chapter}{Notation}\label{sec:notation}

\begin{longtable}{l l p{0.55\textwidth}}
\toprule
& Symbol & Meaning \\
\midrule
General & $\secparam$ & security parameter \\
 & PPT & probabilistic polynomial time \\
 & $\negl(\secparam)$ & a negligible function \\
 & $x \sample S$ & $x$ sampled uniformly from the set $S$ \\
 & $y \sample \algo{A}(x)$ & randomised algorithm $\algo{A}$ on $x$ \\
 & $y \leftarrow \algo{A}(x)$ & deterministic algorithm $\algo{A}$ on $x$ \\
 & $\algo{A}(x; \rho)$ & $\algo{A}$ run on $x$ with randomness $\rho$, so $y \sample \algo{A}(x)$ is $y \leftarrow \algo{A}(x; \rho)$ for uniform $\rho$ \\
 & $\{0,1\}^*$ & bit strings of any length \\
 & $|x|$ & length of $x$ \\
 & $[1,n]$ & the set $\{1, \ldots, n\}$ \\
 & $x \| y$ & concatenation \\
 & $\vec{x}$ & vector \\
 & $\mat{K}$ & matrix \\
\midrule
Security & $\adv$, $\advB$ & adversaries \\
 & $\Adv^{\mathsf{x}}_{\algo{P},\adv}(\secparam)$ & advantage of $\adv$ against property $\mathsf{x}$ of $\algo{P}$ \\
 & $\idfu{}$ & ideal functionality \\
 & $\simu$ & simulator \\
 & $\textsc{Exec}$ & joint output of a real or ideal execution \\
\midrule
Fields and groups & $\Zp$, $\Zp^*$ & integers modulo the prime $p$, and its non-zero elements \\
 & $\group{1}$, $\group{2}$, $\group{T}$ & source and target groups of prime order $p$ \\
 & $\gen$, $\ggen$ & generators of $\group{1}$ and $\group{2}$ \\
 & $\pairing$ & pairing $\group{1} \times \group{2} \rightarrow \group{T}$ \\
\bottomrule
\end{longtable}
