\section{Threshold Structure-Preserving Signatures}\label{sec:schemes}

This section gives threshold variants of the structure-preserving signatures
described in Section~\ref{sec:SPS}, built from the MPC ideal functionalities
of Section~\ref{sec:MPC}~\cite{C:AGHO11,C:KilPanWee15}.

\subsection{Threshold structure-preserving signatures from~\cite{C:AGHO11}}\label{sec:tsps-abe}

We formally describe a threshold structure-preserving signature scheme based
on the structure-preserving scheme of Abe, Groth, Haralambiev, and Ohkubo,
built from the MPC ideal functionalities of
Section~\ref{sec:MPC}~\cite{C:AGHO11}.

\begin{protocolbox}{Threshold structure-preserving signature based
  on~\cite{C:AGHO11}}{tsps-abe}

  \begin{description}

    \item[$(\pk, \llbracket \sk \rrbracket) \sample \algo{KeyGen}(\secparam, l, t,
      n)$ :] On input the security parameter, vector length $l$, corruption
      threshold $t$, and number of signers $n$, the honest signers do the
      following:

      \begin{enumerate}

        \item Issue $l+1$ parallel calls to $\idfu{\algo{RandShare}}$ to
          compute $t$-additive shares $(\llbracket u_1 \rrbracket, \ldots,
          \llbracket u_l\rrbracket, \llbracket v\rrbracket)$.

        \item Call $\idfu{\algo{ExpGenGrp}_1}$ with generator $\gen$ and
          shares $\llbracket v\rrbracket$ to obtain $t$-additive shares
          $\llbracket {V} \rrbracket = {\llbracket v \rrbracket}\gen$, and
          $\idfu{\algo{ExpGenGrp}_2}$ with $\ggen$ and shares $\llbracket
          u_i\rrbracket$ to obtain $t$-additive shares $\llbracket
          \gggen{U}_i\rrbracket = {\llbracket u_i\rrbracket}\ggen$ for $i \in
          [l]$.

        \item Query $\idfu{\algo{CombineGrp}_1}$ with the shares $\llbracket
          {V} \rrbracket$, and $\idfu{\algo{CombineGrp}_2}$ with the shares
          $(\llbracket \gggen{U}_1\rrbracket, \ldots, \llbracket
          \gggen{U}_l\rrbracket)$.

        \item If no $\bot$ occurs, return public key $\pk =
          (\vec{\gggen{U}}, {V})$ with $\vec{\gggen{U}} = (\gggen{U}_1,
          \ldots, \gggen{U}_l)$, with corresponding secret shares
          $\llbracket \sk \rrbracket = (\llbracket u_1 \rrbracket, \ldots,
          \llbracket u_l\rrbracket, \llbracket v \rrbracket)$ distributed to
          the signers.

      \end{enumerate}

    \item[$\llbracket \sigma \rrbracket_{\algo{OffSign}} \sample \algo{OffSign}(\llbracket
      \sk \rrbracket)$ :] On input the secret key shares, the honest signers
      do the following:

      \begin{enumerate}

        \item Invoke $\idfu{\algo{RandShare}}$ to get $t$-additive shares
          $\llbracket r \rrbracket$.

        \item If no $\bot$ occurs, call $\idfu{\algo{Inv}}$ to compute
          $t$-additive shares $\llbracket 1/r\rrbracket$.

        \item Issue, potentially in parallel, $l+1$ calls to
          $\idfu{\algo{Mul}}$ to compute the $t$-additive shares
          $(\llbracket ru_1\rrbracket, \ldots,\llbracket ru_l\rrbracket,
          \llbracket v/r\rrbracket)$.

        \item If the calls to $\idfu{\algo{Inv}}$ and $\idfu{\algo{Mul}}$
          all succeed, the signers hold shares $(\llbracket 1/r\rrbracket,
          \llbracket v/r\rrbracket, \llbracket ru_1\rrbracket, \ldots,
          \llbracket ru_l\rrbracket)$.

        \item Invoke $\idfu{\algo{ExpGenGrp}_1}$ with generator $\gen$ and
          shares $\llbracket r\rrbracket$ to compute $\llbracket
          r\gen\rrbracket$, and $\idfu{\algo{ExpGenGrp}_2}$ with generator
          $\ggen$ and shares $\llbracket 1/r\rrbracket$ and $\llbracket
          v/r\rrbracket$ to compute $\llbracket \gggen{R} \rrbracket =
          \llbracket \tfrac{1}{r}\ggen\rrbracket$ and $\llbracket
          \gggen{S}\rrbracket = \llbracket \tfrac{v}{r}\ggen \rrbracket$.

        \item Conclude the offline phase by storing $(\llbracket
          ru_1\rrbracket, \ldots, \llbracket ru_l\rrbracket, \llbracket
          r\gen\rrbracket, \llbracket \gggen{R}\rrbracket, \llbracket
          \gggen{S}\rrbracket)$.

      \end{enumerate}

    \item[$\llbracket \sigma \rrbracket_{\algo{OnSign}} \sample
      \algo{OnSign}(\vec{{M}})$ :] On input message vector $\vec{{M}}$, the
      honest signers do the following:

      \begin{enumerate}

        \item For each ${M}_i \neq 0$, issue a call to
          $\idfu{\algo{ExpGenGrp}_1}$ with shares $\llbracket u_ir\rrbracket$
          and generator $-{M}_i$ to obtain $t$-additive shares $\llbracket
          -u_ir{M}_i \rrbracket$.

        \item Issue consecutive calls to $\idfu{\algo{MulGrp}_1}$ to compute
          the shares $\llbracket {T}\rrbracket = \llbracket r\gen +
          \sum_{i=1, {M}_i \neq 0}^l {-u_ir{M}_i} \rrbracket$, noting that
          this sum is equivalent to $r\gen - \sum_{i=1}^l u_ir{M}_i$.

      \end{enumerate}

    \item[$\sigma \leftarrow \algo{Combine}(\llbracket \sigma \rrbracket)$ :]
      On input the signature shares, the honest signers do the following:

      \begin{enumerate}

        \item Call $\idfu{\algo{CombineGrp}_2}$ with $\llbracket
          \gggen{R}\rrbracket$ and $\llbracket \gggen{S}\rrbracket$ to
          obtain $\gggen{R} = \tfrac{1}{r}\ggen$ and $\gggen{S} =
          \tfrac{v}{r}\ggen$, and $\idfu{\algo{CombineGrp}_1}$ with shares
          $\llbracket {T}\rrbracket$ to compute ${T} = r(\gen -
          \sum_{i=1}^l u_i{M}_i)$.

        \item If no $\bot$ occurs, return the valid signature $\sigma =
          (\gggen{R}, \gggen{S}, {T})$ on the vector $({M}_1, \ldots,{M}_l)
          \in \group{1}^l$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \algo{Verify}(\pk, \vec{{M}}, \sigma)$ :] On
      input signature $\sigma = (\gggen{R}, \gggen{S}, {T})$, vector
      $({M}_1, \ldots,{M}_l) \in \group{1}^l$, and public key $\pk =
      (\vec{\gggen{U}}, {V})$, do the following:

      \begin{enumerate}

        \item Check whether Equation~\ref{eq:abe-ver} holds, returning 1 if
          so and 0 otherwise.

      \end{enumerate}

  \end{description}

\end{protocolbox}

\subsection{Threshold structure-preserving signatures from~\cite{C:KilPanWee15}}\label{sec:tsps-kiltz}

We now formally describe a threshold structure-preserving signature scheme
based on the structure-preserving scheme of Kiltz, Pan, and Wee, again built
from the MPC ideal functionalities of
Section~\ref{sec:MPC}~\cite{C:KilPanWee15}.

\begin{protocolbox}{Threshold structure-preserving signature based
  on~\cite{C:KilPanWee15}}{tsps-kiltz}

  \begin{description}

    \item[$(\pk, \llbracket \sk \rrbracket) \sample \algo{KeyGen}(\secparam, l, t,
      n)$ :] On input the security parameter, vector length $l$, corruption
      threshold $t$, and number of signers $n$, the honest signers do the
      following.

      \begin{enumerate}

        \item \emph{$\vec{a} = (1, a)^{\mathsf{T}} \in \Zp^2$ and $\vec{b} = (1,
          b)^{\mathsf{T}} \in \Zp^2$.} Issue two calls to
          $\idfu{\algo{RandShare}}$ to compute $t$-additive secret shares $(\llbracket a
          \rrbracket, \llbracket b\rrbracket)$.

        \item \emph{$\mathbf{k} \in \Zp^{(l+1)\times 2}$.} Issue $2(l+1)$
          calls to $\idfu{\algo{RandShare}}$ to get $t$-additive secret shares
          $\llbracket \mathbf{k} \rrbracket = (\llbracket k_{1,1} \rrbracket,
          \ldots, \llbracket k_{l+1,2}\rrbracket)$.

        \item \emph{$\mathbf{c}, \mathbf{d} \in \Zp^{2\times 2}$.} Invoke
          $\idfu{\algo{RandShare}}$ to obtain $t$-additive secret
          shares $(\llbracket c_{1,1} \rrbracket,\ldots, \llbracket c_{2,2}
          \rrbracket, \llbracket d_{1,1}\rrbracket,\ldots, \llbracket
          d_{2,2}\rrbracket)$. If no $\bot$ occurs during randomness
          generation, continue.

        \item \emph{$\vec{u} = \mathbf{k}\vec{a}$.} Invoke
          $\idfu{\algo{Mul}}$ with shares $\llbracket a \rrbracket$ and $\llbracket
          k_{i, 2}\rrbracket$, for $i \in [l+1]$, to compute $t$-additive
          secret shares $(\llbracket k_{1, 2}a \rrbracket,\ldots, \llbracket
          k_{l+1,2}a \rrbracket)$. If no $\bot$ occurs, call $\idfu{\algo{Add}}$
          with shares $\llbracket k_{i, 1}\rrbracket$ and $\llbracket k_{i,
          2}a \rrbracket$ to obtain $t$-additive secret shares $(\llbracket
          u_1 \rrbracket,\ldots, \llbracket u_{l+1} \rrbracket)$ with $u_i =
          k_{i, 1}+ k_{i, 2}a$ for all $i \in [l+1]$.

        \item \emph{$\vec{v} = \mathbf{c}\vec{a}$.} Invoke
          $\idfu{\algo{Mul}}$ with shares $\llbracket a \rrbracket$ and $\llbracket
          c_{1, 2}\rrbracket$, and with $\llbracket a \rrbracket$ and
          $\llbracket c_{2, 2}\rrbracket$, to compute $t$-additive secret
          shares $(\llbracket c_{1, 2}a \rrbracket,\llbracket c_{2,2}a
          \rrbracket)$. If no $\bot$ occurs, call $\idfu{\algo{Add}}$ with
          shares $\llbracket c_{1, 1}\rrbracket$ and $\llbracket c_{1, 2}a
          \rrbracket$, and with $\llbracket c_{2, 1}\rrbracket$ and
          $\llbracket c_{2, 2}a \rrbracket$, to obtain $t$-additive secret
          shares $(\llbracket v_1 \rrbracket, \llbracket v_2 \rrbracket)$ with
          $v_1 = c_{1, 1}+ c_{1, 2}a$ and $v_2 = c_{2, 1}+ c_{2, 2}a$.

        \item \emph{$\vec{w} = \mathbf{d}\vec{a}$.} Follow the
          same steps as for $\vec{v}$, with $\mathbf{c}$ replaced by $\mathbf{d}$
          and $\vec{v}$ replaced by $\vec{w}$.

        \item \emph{$\vec{x} = \vec{b}^{\mathsf{T}}\mathbf{c}$.} Invoke
          $\idfu{\algo{Mul}}$ with shares $\llbracket b \rrbracket$ and
          $\llbracket c_{2, 1}\rrbracket$, and with $\llbracket b \rrbracket$
          and $\llbracket c_{2, 2}\rrbracket$, to get $t$-additive secret
          shares $(\llbracket bc_{2, 1} \rrbracket,\llbracket bc_{2,2}
          \rrbracket)$. If no $\bot$ occurs, call $\idfu{\algo{Add}}$ with
          shares $\llbracket c_{1, 1}\rrbracket$ and $\llbracket bc_{2, 1}
          \rrbracket$, and with $\llbracket c_{1, 2}\rrbracket$ and
          $\llbracket bc_{2, 2} \rrbracket$, to obtain $t$-additive secret
          shares $(\llbracket x_1 \rrbracket, \llbracket x_2 \rrbracket)$
          with $x_1 = c_{1, 1}+ bc_{2, 1}$ and $x_2 = c_{1, 2}+ bc_{2, 2}$.

        \item \emph{$\vec{y} = \vec{b}^{\mathsf{T}}\mathbf{d}$.} Follow the
          same steps as for $\vec{x}$, with $\mathbf{c}$ replaced by
          $\mathbf{d}$ and $\vec{x}$ replaced by $\vec{y}$.

        \item \emph{$(k_{1, 1}\gen, k_{1,2}\gen)$, $\vec{{B}} =
          (\gen, b\gen)$, $\vec{{X}} = ({X}_1, {X}_2) =
          (x_1\gen, x_2\gen)$, $\vec{{Y}} = ({Y}_1, {Y}_2) =
          (y_1\gen, y_2\gen)$.} Invoke
          $\idfu{\algo{ExpGenGrp}_1}$ with generator $\gen$ and shares $\llbracket
          k_{1, 1}\rrbracket$, $\llbracket k_{1, 2} \rrbracket$, $\llbracket b
          \rrbracket$, $\llbracket x_1\rrbracket$, $\llbracket
          x_2\rrbracket$, $\llbracket y_1\rrbracket$, and $\llbracket
          y_2\rrbracket$, to compute the corresponding $t$-additive shares.

        \item \emph{$\vec{\gggen{A}} = (\ggen, a\ggen)$, $\vec{\gggen{U}} =
          (u_1\ggen, \ldots, u_{l+1}\ggen)$, $\vec{\gggen{V}} = (v_1\ggen,
          v_2\ggen)$, $\vec{\gggen{W}} = (w_1\ggen, w_2\ggen)$.} Invoke
          $\idfu{\algo{ExpGenGrp}_2}$ with generator $\ggen$ and
          shares $\llbracket a \rrbracket$, $\llbracket u_1\rrbracket \ldots
          \llbracket u_{l+1}\rrbracket$, $\llbracket v_1\rrbracket$,
          $\llbracket v_2\rrbracket$, $\llbracket w_1\rrbracket$, and
          $\llbracket w_2\rrbracket$, to compute the corresponding
          $t$-additive shares.

        \item Call $\idfu{\algo{CombineGrp}_2}$ to compute and
          publish public key $\pk =(\vec{\gggen{A}}, \vec{\gggen{U}},
          \vec{\gggen{V}},\vec{\gggen{W}})$. If no $\bot$ occurs, proceed to
          the next step.

        \item Conclude key generation by storing the
          shares $\llbracket \sk \rrbracket = (\llbracket \mathbf{k}\rrbracket,
          \llbracket {B}\rrbracket, \llbracket \vec{{X}} \rrbracket,
          \llbracket \vec{{Y}} \rrbracket)$, together with the pair
          $\langle \llbracket k_{1, 1}\gen \rrbracket,\llbracket k_{1,
          2}\gen \rrbracket \rangle$.

      \end{enumerate}

    \item[$\llbracket \sigma \rrbracket_{\algo{OffSign}} \sample \algo{OffSign}(\llbracket
      \sk \rrbracket)$ :] On input the secret key shares, the honest signers
      do the following:

      \begin{enumerate}

        \item Invoke $\idfu{\algo{RandShare}}$ to get $t$-additive shares
          $\llbracket r \rrbracket$ and $\llbracket s \rrbracket$.

        \item If no $\bot$ occurs, call $\idfu{\algo{Mul}}$ with shares
          $\llbracket r \rrbracket$ and $\llbracket s \rrbracket$ to compute
          $t$-additive shares $\llbracket rs \rrbracket$.

        \item If this succeeds, invoke $\idfu{\algo{ExpGrp}_1}$ to compute
          $t$-additive shares $\llbracket r{B}\rrbracket$, $\llbracket
          rs{B}\rrbracket$, $\llbracket r\vec{{X}}\rrbracket$, and
          $\llbracket rs\vec{{Y}}\rrbracket$.

        \item If $\idfu{\algo{ExpGrp}_1}$ does not abort, invoke
          $\idfu{\algo{ExpGenGrp}_1}$ to obtain $t$-additive shares $\llbracket
          r\gen\rrbracket$ and $\llbracket rs\gen\rrbracket$, and
          $\idfu{\algo{ExpGenGrp}_2}$ to get $t$-additive shares $\llbracket
          s\ggen\rrbracket$.

        \item If no $\bot$ occurs, call $\idfu{\algo{MulGrp}_1}$ to receive
          $t$-additive shares $\llbracket \vec{{T}}^{\star} \rrbracket =
          \llbracket r\vec{{X}} + rs\vec{{Y}}\rrbracket= \llbracket
          r\vec{{X}}\rrbracket +\llbracket rs\vec{{Y}}\rrbracket$.

        \item Conclude the offline phase by storing $(\llbracket
          s\ggen\rrbracket, \llbracket r\gen\rrbracket, \llbracket
          rs\gen\rrbracket, \llbracket r{B}\rrbracket, \llbracket
          rs{B}\rrbracket, \llbracket \vec{{T}}^{\star}\rrbracket)$.

      \end{enumerate}

    \item[$\llbracket \sigma \rrbracket_{\algo{OnSign}} \sample
      \algo{OnSign}(\vec{{M}})$ :] On input message vector $\vec{{M}}$, the
      honest signers do the following:

      \begin{enumerate}

        \item For each ${M}_i \neq 0$, issue a call to
          $\idfu{\algo{ExpGenGrp}_1}$ with shares $\llbracket k_{i,
          1}\rrbracket$ and $\llbracket k_{i, 2} \rrbracket$ to obtain
          $t$-additive shares $\llbracket k_{i, 1} {M}_i\rrbracket =
          \llbracket k_{i, 1}\rrbracket {M}_i$ and $\llbracket k_{i+1, 2}
          {M}_i\rrbracket = \llbracket k_{i+1, 2}\rrbracket{M}_i$.

        \item Issue consecutive calls to $\idfu{\algo{MulGrp}_1}$ to compute
          the $t$-additive shares $\llbracket(\gen,
          \vec{M})\mathbf{k}\rrbracket$, comprising:
          %
          \begin{equation*}
          %
            \left(\llbracket k_{1, 1}\gen \rrbracket + \sum_{i=1, {M}_i \neq
            0}^{l+1} \llbracket k_{i+1, 1}{M}_i\rrbracket , \llbracket k_{1,
            2}\gen\rrbracket + \sum_{i=1, {M}_i \neq 0}^{l+1} \llbracket
            k_{i+1, 2}{M}_i\rrbracket\right).
          %
          \end{equation*}

        \item Issue one final call to $\idfu{\algo{MulGrp}_1}$ to
          compute the shares $\llbracket \vec{{T}}\rrbracket = \llbracket
          (\gen, \vec{M})\mathbf{k} + \vec{{T}}^{\star}\rrbracket = \llbracket
          (\gen, \vec{M})\mathbf{k} \rrbracket + \llbracket
          \vec{{T}}^{\star}\rrbracket$.

      \end{enumerate}

    \item[$\sigma \leftarrow \algo{Combine}(\llbracket \sigma \rrbracket)$ :]
      On input the signature shares, the honest signers do the following:

      \begin{enumerate}

        \item Query $\idfu{\algo{CombineGrp}_1}$ to compute group elements
          $\vec{{R}} = (r\gen, r{B})$ and $\vec{{S}} = (rs\gen,
          rs{B})$, and $\idfu{\algo{CombineGrp}_2}$ to compute $\gggen{Q} =
          s\ggen$.

        \item Invoke $\idfu{\algo{CombineGrp}_1}$ to compute $\vec{{T}}=
          (\gen, \vec{{M}})\mathbf{k} + \vec{{T}^{\star}}$.

        \item If no $\bot$ occurs, return the valid signature $\sigma =
          (\gggen{Q}, \vec{{R}}, \vec{{S}}, \vec{{T}})$ on vector $\vec{{M}}
          = ({M}_1, \ldots, {M}_l)$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \algo{Verify}(\pk, \vec{{M}}, \sigma)$ :] On
      input signature $\sigma = (\gggen{Q}, \vec{{R}}, \vec{{S}},
      \vec{{T}})$, vector $({M}_1, \ldots,{M}_l) \in \group{1}^l$, and
      public key $\pk =(\vec{\gggen{A}}, \vec{\gggen{U}},
      \vec{\gggen{V}},\vec{\gggen{W}})$, do the following:

      \begin{enumerate}

        \item Check whether Equation~\ref{eq:kiltz-ver} holds, returning 1
          if so and 0 otherwise.

      \end{enumerate}

  \end{description}

\end{protocolbox}

Section~\ref{sec:tsps-security} formalises the security notion for threshold
structure-preserving signatures, adp-TS-UF-1, and proves that both
constructions above satisfy it.
