\section{Threshold Structure-Preserving Signatures}\label{sec:tsps-instantiation}

This section turns the general strategy of Section~\ref{sec:tsps-construction}
into full descriptions of two threshold structure-preserving signature
schemes, built from the MPC ideal functionalities of
Section~\ref{sec:tsps-definitions}~\cite{C:AGHO11,C:KilPanWee15}. Realising
this strategy concretely requires two things beyond the generic MPC
functionalities over $\Zp$: computing on secret-shared group elements, via
the functionalities over $\G$, and structuring each scheme so that the
online phase, run once the messages to be signed are known, involves no
communication between signers at all. Both schemes below are designed with
this non-interactive online phase in mind, with $\algo{OffSign}$ carrying
essentially all of the interactive computation.

\subsection{Threshold structure-preserving signatures from~\cite{C:AGHO11}}\label{sec:tsps-abe}

We formally describe a threshold structure-preserving signature scheme based
on the structure-preserving scheme of Abe, Groth, Haralambiev, and Ohkubo,
built from the MPC ideal functionalities of
Section~\ref{sec:MPC}~\cite{C:AGHO11}.

\begin{protocolbox}{Threshold structure-preserving signature based
  on~\cite{C:AGHO11}}{tsps-abe}

  \begin{description}

    \item[$(\pk, \lbrack \sk\rbrack ) \sample \algo{KeyGen}(\secparam, l, t,
      n)$ :] On input the security parameter, vector length $l$, corruption
      threshold $t$, and number of signers $n$, the honest signers do the
      following:

      \begin{enumerate}

        \item Issue $l+1$ parallel calls to $\idfu{\algo{RandShare}}$ to
          compute $t$-additive shares $(\lbrack u_1\rbrack , \ldots,
          \lbrack u_l\rbrack , \lbrack v\rbrack )$.

        \item Call $\idfu{\algo{ExpGenGrp}_1}$ with generator $\gen$ and
          shares $\lbrack v\rbrack $ to obtain $t$-additive shares
          $\lbrack {V}\rbrack  = {\lbrack v\rbrack }\gen$, and
          $\idfu{\algo{ExpGenGrp}_2}$ with $\ggen$ and shares $\lbrack 
          u_i\rbrack $ to obtain $t$-additive shares $\lbrack 
          \gggen{U}_i\rbrack  = {\lbrack u_i\rbrack }\ggen$ for $i \in [l]$.

        \item Query $\idfu{\algo{CombineGrp}_1}$ with the shares $\lbrack 
          {V}\rbrack $, and $\idfu{\algo{CombineGrp}_2}$ with the shares
          $(\lbrack \gggen{U}_1\rbrack , \ldots, \lbrack 
          \gggen{U}_l\rbrack )$.

        \item If no $\bot$ occurs, return public key $\pk =
          (\vec{\gggen{U}}, {V})$ with $\vec{\gggen{U}} = (\gggen{U}_1,
          \ldots, \gggen{U}_l)$, with corresponding secret shares
          $\lbrack \sk\rbrack  = (\lbrack u_1\rbrack , \ldots,
          \lbrack u_l\rbrack , \lbrack v\rbrack )$ distributed to
          the signers.

      \end{enumerate}

    \item[$\lbrack \signature\rbrack _{\algo{OffSign}} \sample \algo{OffSign}(\lbrack 
      \sk\rbrack )$ :] On input the secret key shares, the honest signers
      do the following:

      \begin{enumerate}

        \item Invoke $\idfu{\algo{RandShare}}$ to get $t$-additive shares
          $\lbrack r\rbrack $.

        \item If no $\bot$ occurs, call $\idfu{\algo{Inv}}$ to compute
          $t$-additive shares $\lbrack 1/r\rbrack $.

        \item Issue, potentially in parallel, $l+1$ calls to
          $\idfu{\algo{Mul}}$ to compute the $t$-additive shares
          $(\lbrack ru_1\rbrack , \ldots,\lbrack ru_l\rbrack ,
          \lbrack v/r\rbrack )$.

        \item If the calls to $\idfu{\algo{Inv}}$ and $\idfu{\algo{Mul}}$
          all succeed, the signers hold shares $(\lbrack 1/r\rbrack ,
          \lbrack v/r\rbrack , \lbrack ru_1\rbrack , \ldots,
          \lbrack ru_l\rbrack )$.

        \item Invoke $\idfu{\algo{ExpGenGrp}_1}$ with generator $\gen$ and
          shares $\lbrack r\rbrack $ to compute $\lbrack 
          r\gen\rbrack $, and $\idfu{\algo{ExpGenGrp}_2}$ with generator
          $\ggen$ and shares $\lbrack 1/r\rbrack $ and $\lbrack 
          v/r\rbrack $ to compute $\lbrack \gggen{R}\rbrack  =
          \lbrack \tfrac{1}{r}\ggen\rbrack $ and $\lbrack 
          \gggen{S}\rbrack  = \lbrack \tfrac{v}{r}\ggen\rbrack $.

        \item Conclude the offline phase by storing $(\lbrack 
          ru_1\rbrack , \ldots, \lbrack ru_l\rbrack , \lbrack 
          r\gen\rbrack , \lbrack \gggen{R}\rbrack , \lbrack 
          \gggen{S}\rbrack )$.

      \end{enumerate}

    \item[$\lbrack \signature\rbrack _{\algo{OnSign}} \sample
      \algo{OnSign}(\vec{{M}})$ :] On input message vector $\vec{{M}}$, the
      honest signers do the following:

      \begin{enumerate}

        \item For each ${M}_i \neq 0$, issue a call to
          $\idfu{\algo{ExpGenGrp}_1}$ with shares $\lbrack u_ir\rbrack $
          and generator $-{M}_i$ to obtain $t$-additive shares $\lbrack 
          -u_ir{M}_i\rbrack $.

        \item Issue consecutive calls to $\idfu{\algo{MulGrp}_1}$ to compute
          the shares $\lbrack {T}\rbrack  = \lbrack r\gen +
          \sum_{i=1, {M}_i \neq 0}^l {-u_ir{M}_i}\rbrack $, noting that
          this sum is equivalent to $r\gen - \sum_{i=1}^l u_ir{M}_i$.

      \end{enumerate}

    \item[$\signature \leftarrow \algo{Combine}(\lbrack \signature\rbrack )$ :]
      On input the signature shares, the honest signers do the following:

      \begin{enumerate}

        \item Call $\idfu{\algo{CombineGrp}_2}$ with $\lbrack 
          \gggen{R}\rbrack $ and $\lbrack \gggen{S}\rbrack $ to
          obtain $\gggen{R} = \tfrac{1}{r}\ggen$ and $\gggen{S} =
          \tfrac{v}{r}\ggen$, and $\idfu{\algo{CombineGrp}_1}$ with shares
          $\lbrack {T}\rbrack $ to compute ${T} = r(\gen -
          \sum_{i=1}^l u_i{M}_i)$.

        \item If no $\bot$ occurs, return the valid signature $\signature =
          (\gggen{R}, \gggen{S}, {T})$ on the vector $({M}_1, \ldots,{M}_l)
          \in \group{1}^l$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \SPSverify(\pk, \vec{{M}}, \signature)$ :] On
      input signature $\signature = (\gggen{R}, \gggen{S}, {T})$, vector
      $({M}_1, \ldots,{M}_l) \in \group{1}^l$, and public key $\pk =
      (\vec{\gggen{U}}, {V})$, do the following:

      \begin{enumerate}

        \item Check whether Equation~\ref{eq:abe-ver} holds, returning 1 if
          so and 0 otherwise.

      \end{enumerate}

  \end{description}

\end{protocolbox}

\subsection{Threshold structure-preserving signatures from~\cite{C:KilPanWee15}}\label{sec:tsps-kiltz}

We now formally describe a threshold structure-preserving signature scheme
based on the structure-preserving scheme of Kiltz, Pan, and Wee, again built
from the MPC ideal functionalities of
Section~\ref{sec:MPC}~\cite{C:KilPanWee15}.

\begin{protocolbox}{Threshold structure-preserving signature based
  on~\cite{C:KilPanWee15}}{tsps-kiltz}

  \begin{description}

    \item[$(\pk, \lbrack \sk\rbrack ) \sample \algo{KeyGen}(\secparam, l, t,
      n)$ :] On input the security parameter, vector length $l$, corruption
      threshold $t$, and number of signers $n$, the honest signers do the
      following.

      \begin{enumerate}

        \item \emph{$\vec{a} = (1, a)^{\mathsf{T}} \in \Zp^2$ and $\vec{b} = (1,
          b)^{\mathsf{T}} \in \Zp^2$.} Issue two calls to
          $\idfu{\algo{RandShare}}$ to compute $t$-additive secret shares $(\lbrack a
\rbrack , \lbrack b\rbrack )$.

        \item \emph{$\mathbf{k} \in \Zp^{(l+1)\times 2}$.} Issue $2(l+1)$
          calls to $\idfu{\algo{RandShare}}$ to get $t$-additive secret shares
          $\lbrack \mathbf{k}\rbrack  = (\lbrack k_{1,1}\rbrack ,
          \ldots, \lbrack k_{l+1,2}\rbrack )$.

        \item \emph{$\mathbf{c}, \mathbf{d} \in \Zp^{2\times 2}$.} Invoke
          $\idfu{\algo{RandShare}}$ to obtain $t$-additive secret
          shares $(\lbrack c_{1,1}\rbrack ,\ldots, \lbrack c_{2,2}
\rbrack , \lbrack d_{1,1}\rbrack ,\ldots, \lbrack 
          d_{2,2}\rbrack )$. If no $\bot$ occurs during randomness
          generation, continue.

        \item \emph{$\vec{u} = \mathbf{k}\vec{a}$.} Invoke
          $\idfu{\algo{Mul}}$ with shares $\lbrack a\rbrack $ and $\lbrack 
          k_{i, 2}\rbrack $, for $i \in [l+1]$, to compute $t$-additive
          secret shares $(\lbrack k_{1, 2}a\rbrack ,\ldots, \lbrack 
          k_{l+1,2}a\rbrack )$. If no $\bot$ occurs, call $\idfu{\algo{Add}}$
          with shares $\lbrack k_{i, 1}\rbrack $ and $\lbrack k_{i,
          2}a\rbrack $ to obtain $t$-additive secret shares $(\lbrack 
          u_1\rbrack ,\ldots, \lbrack u_{l+1}\rbrack )$ with $u_i =
          k_{i, 1}+ k_{i, 2}a$ for all $i \in [l+1]$.

        \item \emph{$\vec{v} = \mathbf{c}\vec{a}$.} Invoke
          $\idfu{\algo{Mul}}$ with shares $\lbrack a\rbrack $ and $\lbrack 
          c_{1, 2}\rbrack $, and with $\lbrack a\rbrack $ and
          $\lbrack c_{2, 2}\rbrack $, to compute $t$-additive secret
          shares $(\lbrack c_{1, 2}a\rbrack ,\lbrack c_{2,2}a
\rbrack )$. If no $\bot$ occurs, call $\idfu{\algo{Add}}$ with
          shares $\lbrack c_{1, 1}\rbrack $ and $\lbrack c_{1, 2}a
\rbrack $, and with $\lbrack c_{2, 1}\rbrack $ and
          $\lbrack c_{2, 2}a\rbrack $, to obtain $t$-additive secret
          shares $(\lbrack v_1\rbrack , \lbrack v_2\rbrack )$ with
          $v_1 = c_{1, 1}+ c_{1, 2}a$ and $v_2 = c_{2, 1}+ c_{2, 2}a$.

        \item \emph{$\vec{w} = \mathbf{d}\vec{a}$.} Follow the
          same steps as for $\vec{v}$, with $\mathbf{c}$ replaced by $\mathbf{d}$
          and $\vec{v}$ replaced by $\vec{w}$.

        \item \emph{$\vec{x} = \vec{b}^{\mathsf{T}}\mathbf{c}$.} Invoke
          $\idfu{\algo{Mul}}$ with shares $\lbrack b\rbrack $ and
          $\lbrack c_{2, 1}\rbrack $, and with $\lbrack b\rbrack $
          and $\lbrack c_{2, 2}\rbrack $, to get $t$-additive secret
          shares $(\lbrack bc_{2, 1}\rbrack ,\lbrack bc_{2,2}
\rbrack )$. If no $\bot$ occurs, call $\idfu{\algo{Add}}$ with
          shares $\lbrack c_{1, 1}\rbrack $ and $\lbrack bc_{2, 1}
\rbrack $, and with $\lbrack c_{1, 2}\rbrack $ and
          $\lbrack bc_{2, 2}\rbrack $, to obtain $t$-additive secret
          shares $(\lbrack x_1\rbrack , \lbrack x_2\rbrack )$
          with $x_1 = c_{1, 1}+ bc_{2, 1}$ and $x_2 = c_{1, 2}+ bc_{2, 2}$.

        \item \emph{$\vec{y} = \vec{b}^{\mathsf{T}}\mathbf{d}$.} Follow the
          same steps as for $\vec{x}$, with $\mathbf{c}$ replaced by
          $\mathbf{d}$ and $\vec{x}$ replaced by $\vec{y}$.

        \item \emph{$(k_{1, 1}\gen, k_{1,2}\gen)$, $\vec{{B}} =
          (\gen, b\gen)$, $\vec{{X}} = ({X}_1, {X}_2) =
          (x_1\gen, x_2\gen)$, $\vec{{Y}} = ({Y}_1, {Y}_2) =
          (y_1\gen, y_2\gen)$.} Invoke
          $\idfu{\algo{ExpGenGrp}_1}$ with generator $\gen$ and shares $\lbrack 
          k_{1, 1}\rbrack $, $\lbrack k_{1, 2}\rbrack $, $\lbrack b
\rbrack $, $\lbrack x_1\rbrack $, $\lbrack 
          x_2\rbrack $, $\lbrack y_1\rbrack $, and $\lbrack 
          y_2\rbrack $, to compute the corresponding $t$-additive shares.

        \item \emph{$\vec{\gggen{A}} = (\ggen, a\ggen)$, $\vec{\gggen{U}} =
          (u_1\ggen, \ldots, u_{l+1}\ggen)$, $\vec{\gggen{V}} = (v_1\ggen,
          v_2\ggen)$, $\vec{\gggen{W}} = (w_1\ggen, w_2\ggen)$.} Invoke
          $\idfu{\algo{ExpGenGrp}_2}$ with generator $\ggen$ and
          shares $\lbrack a\rbrack $, $\lbrack u_1\rbrack  \ldots
          \lbrack u_{l+1}\rbrack $, $\lbrack v_1\rbrack $,
          $\lbrack v_2\rbrack $, $\lbrack w_1\rbrack $, and
          $\lbrack w_2\rbrack $, to compute the corresponding
          $t$-additive shares.

        \item Call $\idfu{\algo{CombineGrp}_2}$ to compute and
          publish public key $\pk =(\vec{\gggen{A}}, \vec{\gggen{U}},
          \vec{\gggen{V}},\vec{\gggen{W}})$. If no $\bot$ occurs, proceed to
          the next step.

        \item Conclude key generation by storing the
          shares $\lbrack \sk\rbrack  = (\lbrack \mathbf{k}\rbrack ,
          \lbrack {B}\rbrack , \lbrack \vec{{X}}\rbrack ,
          \lbrack \vec{{Y}}\rbrack )$, together with the pair
          $\langle \lbrack k_{1, 1}\gen\rbrack ,\lbrack k_{1,
          2}\gen\rbrack  \rangle$.

      \end{enumerate}

    \item[$\lbrack \signature\rbrack _{\algo{OffSign}} \sample \algo{OffSign}(\lbrack 
      \sk\rbrack )$ :] On input the secret key shares, the honest signers
      do the following:

      \begin{enumerate}

        \item Invoke $\idfu{\algo{RandShare}}$ to get $t$-additive shares
          $\lbrack r\rbrack $ and $\lbrack s\rbrack $.

        \item If no $\bot$ occurs, call $\idfu{\algo{Mul}}$ with shares
          $\lbrack r\rbrack $ and $\lbrack s\rbrack $ to compute
          $t$-additive shares $\lbrack rs\rbrack $.

        \item If this succeeds, invoke $\idfu{\algo{ExpGrp}_1}$ to compute
          $t$-additive shares $\lbrack r{B}\rbrack $, $\lbrack 
          rs{B}\rbrack $, $\lbrack r\vec{{X}}\rbrack $, and
          $\lbrack rs\vec{{Y}}\rbrack $.

        \item If $\idfu{\algo{ExpGrp}_1}$ does not abort, invoke
          $\idfu{\algo{ExpGenGrp}_1}$ to obtain $t$-additive shares $\lbrack 
          r\gen\rbrack $ and $\lbrack rs\gen\rbrack $, and
          $\idfu{\algo{ExpGenGrp}_2}$ to get $t$-additive shares $\lbrack 
          s\ggen\rbrack $.

        \item If no $\bot$ occurs, call $\idfu{\algo{MulGrp}_1}$ to receive
          $t$-additive shares $\lbrack \vec{{T}}^{\star}\rbrack  =
          \lbrack r\vec{{X}} + rs\vec{{Y}}\rbrack = \lbrack 
          r\vec{{X}}\rbrack  +\lbrack rs\vec{{Y}}\rbrack $.

        \item Conclude the offline phase by storing $(\lbrack 
          s\ggen\rbrack , \lbrack r\gen\rbrack , \lbrack 
          rs\gen\rbrack , \lbrack r{B}\rbrack , \lbrack 
          rs{B}\rbrack , \lbrack \vec{{T}}^{\star}\rbrack )$.

      \end{enumerate}

    \item[$\lbrack \signature\rbrack _{\algo{OnSign}} \sample
      \algo{OnSign}(\vec{{M}})$ :] On input message vector $\vec{{M}}$, the
      honest signers do the following:

      \begin{enumerate}

        \item For each ${M}_i \neq 0$, issue a call to
          $\idfu{\algo{ExpGenGrp}_1}$ with shares $\lbrack k_{i,
          1}\rbrack $ and $\lbrack k_{i, 2}\rbrack $ to obtain
          $t$-additive shares $\lbrack k_{i, 1} {M}_i\rbrack  =
          \lbrack k_{i, 1}\rbrack  {M}_i$ and $\lbrack k_{i+1, 2}
          {M}_i\rbrack  = \lbrack k_{i+1, 2}\rbrack {M}_i$.

        \item Issue consecutive calls to $\idfu{\algo{MulGrp}_1}$ to compute
          the $t$-additive shares $\lbrack (\gen,
          \vec{M})\mathbf{k}\rbrack $, comprising:
          %
          \begin{equation*}
          %
            \left(\lbrack k_{1, 1}\gen\rbrack  + \sum_{i=1, {M}_i \neq
            0}^{l+1} \lbrack k_{i+1, 1}{M}_i\rbrack  , \lbrack k_{1,
            2}\gen\rbrack  + \sum_{i=1, {M}_i \neq 0}^{l+1} \lbrack 
            k_{i+1, 2}{M}_i\rbrack \right).
          %
          \end{equation*}

        \item Issue one final call to $\idfu{\algo{MulGrp}_1}$ to
          compute the shares $\lbrack \vec{{T}}\rbrack  = \lbrack 
          (\gen, \vec{M})\mathbf{k} + \vec{{T}}^{\star}\rbrack  = \lbrack 
          (\gen, \vec{M})\mathbf{k}\rbrack  + \lbrack 
          \vec{{T}}^{\star}\rbrack $.

      \end{enumerate}

    \item[$\signature \leftarrow \algo{Combine}(\lbrack \signature\rbrack )$ :]
      On input the signature shares, the honest signers do the following:

      \begin{enumerate}

        \item Query $\idfu{\algo{CombineGrp}_1}$ to compute group elements
          $\vec{{R}} = (r\gen, r{B})$ and $\vec{{S}} = (rs\gen,
          rs{B})$, and $\idfu{\algo{CombineGrp}_2}$ to compute $\gggen{Q} =
          s\ggen$.

        \item Invoke $\idfu{\algo{CombineGrp}_1}$ to compute $\vec{{T}}=
          (\gen, \vec{{M}})\mathbf{k} + \vec{{T}^{\star}}$.

        \item If no $\bot$ occurs, return the valid signature $\signature =
          (\gggen{Q}, \vec{{R}}, \vec{{S}}, \vec{{T}})$ on vector $\vec{{M}}
          = ({M}_1, \ldots, {M}_l)$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \SPSverify(\pk, \vec{{M}}, \signature)$ :] On
      input signature $\signature = (\gggen{Q}, \vec{{R}}, \vec{{S}},
      \vec{{T}})$, vector $({M}_1, \ldots,{M}_l) \in \group{1}^l$, and
      public key $\pk =(\vec{\gggen{A}}, \vec{\gggen{U}},
      \vec{\gggen{V}},\vec{\gggen{W}})$, do the following:

      \begin{enumerate}

        \item Check whether Equation~\ref{eq:kiltz-ver} holds, returning 1
          if so and 0 otherwise.

      \end{enumerate}

  \end{description}

\end{protocolbox}

Section~\ref{sec:tsps-security} formalises the security notion for threshold
structure-preserving signatures, adp-TS-UF-1, and proves that both
constructions above satisfy it.
