Throughout, we use hash functions $\hash_p$ into $\Zp$ and
$\hash_{\group{1}}$ into $\group{1}$. The commitment scheme is the Pedersen
commitment scheme, with key $\ck = (\hgen_1, \hgen) \sample
\ComGen(\secparam)$, so that a pseudonym is $\com = \hgen_1^{\algo{uid}}
\hgen^{s}$. The BBS+ generators $\gen_0, \ldots, \gen_3$ and the
generators $P_0, \ldots, P_\ell$ of the central bank's signing on committed
inputs are separate from $\ck$. Every generator is obtained as
$\hash_{\group{1}}$ of its own fixed string, so that no participant knows the
discrete logarithm relations among them.

\subsection{Instantiating $\JPOK$ with BBS+}\label{sec:joint-proof}

A user's credential is a BBS+ signature $\varphi = (e, r, E)$ of
\Cref{prim:bbs} from the registration authority on the pair $(\algo{uid},
\sk)$, with $E$ as follows:
%
\begin{equation*}
  E = (\gen_0 \gen_1^{\algo{uid}} \gen_2^{\sk} \gen_3^{r})^{1/(z+e)},
\end{equation*}
%
where $z$ is the
registration authority's secret key and $\ggen_Z = \ggen^{z}$ its public key.
Every payment and deposit proves knowledge of such a credential, together with
the opening of the payer's pseudonym, without revealing either.

Hesse, Singh and Sorniotti split a proof of knowledge of a BBS+ signature
between two provers: a card, which holds one of the signed messages, and a
holder, which holds the other messages and the signature. \Cref{prim:hss-joint-proof}
gives their protocol.

\begin{primitivebox}{The joint proof of knowledge of a BBS+ signature of Hesse, Singh and Sorniotti~\cite{USENIX:HesSinSor23}}{hss-joint-proof}

  The card and the holder share a PRF key $\key$, and $\varphi = (e, r, E)$ is a
BBS+ signature on $(\msg_1, \ldots, \msg_n)$ under $\pk = (\gen_0, \ldots,
\gen_{n+1}, \ggen, \ggen_Z)$. The card holds $\msg_1 \neq 0$. The holder holds
$Q = \gen_1^{\msg_1}$, the messages $(\msg_i)_{i=2}^{n}$ and $\varphi$. The
verifier holds disclosed values $(v_i)_{i \in V}$ for a set $V \subseteq \{2,
\ldots, n\}$, and $H = \{2, \ldots, n\} \setminus V$. We write
$\SOK\{(\vec{\alpha}) : \ldots\}(n)$ for a signature of knowledge on $n$ of
exponents $\vec{\alpha}$ that satisfy the given equations.

  \begin{description}

    \item[$(D, \pi_C) \sample \algo{Prove}_C(\key, \msg_1, n_V, n)$ :] Run by
the card. Compute $t \leftarrow \PRFeval(\pp, \key, n)$ and $D =
\gen_1^{\msg_1} \gen_{n+1}^{t}$, and output $(D, \pi_C)$ with $\pi_C$ as
follows:
      %
      \begin{equation*}
      %
        \pi_C \sample \SOK\{(\alpha, \beta) : \gen_1^{\alpha} \gen_{n+1}^{\beta}
= D\}(n_V).
      %
      \end{equation*}

    \item[$\Phi_H \sample \algo{Prove}_H(\key, Q, (\msg_i)_{i=2}^{n}, \varphi,
(v_i)_{i \in V}, n, D, n_V)$ :] Run by the holder. Do the following:

      \begin{enumerate}

	\item Output $\perp$ if $\msg_i \neq v_i$ for some $i \in V$.

	\item Compute $t \leftarrow \PRFeval(\pp, \key, n)$, and output $\perp$ if
$Q \gen_{n+1}^{t} \neq D$.

	\item Sample $\varrho_1 \sample \Zp^*$ and $\varrho_2 \sample \Zp$, set
$\varrho_3 = 1/\varrho_1$ and $r' = r - \varrho_2 \varrho_3 - t$, and compute
the following:
          %
          \begin{equation*}
          %
            b = \gen_0 \gen_{n+1}^{r} Q \prod_{i=2}^{n} \gen_i^{\msg_i}, \quad
E' = E^{\varrho_1}, \quad \bar{E} = E'^{-e} b^{\varrho_1}, \quad d =
b^{\varrho_1} \gen_{n+1}^{-\varrho_2}.
          %
          \end{equation*}

	\item Compute the following:
          %
          \begin{multline*}
          %
            \pi_H \sample \SOK\bigl\{(e, r', \varrho_2, \varrho_3, (\msg_i)_{i
\in H}) : E'^{-e} \gen_{n+1}^{\varrho_2} = \bar{E}/d \\
          %
            \wedge\ d^{-\varrho_3} \gen_{n+1}^{r'} \prod_{i \in H}
\gen_i^{\msg_i} = \gen_0^{-1} D^{-1} \prod_{i \in V} \gen_i^{-v_i}\bigr\}(n_V).
          %
          \end{multline*}

	\item Output $\Phi_H = (E', \bar{E}, d, \pi_H)$.

      \end{enumerate}

    \item[Protocol :] The holder, the verifier and the card interact as
follows:

      \begin{enumerate}

	\item The holder samples $n_H \sample \{0,1\}^{\secparam}$ and sends it to
the verifier.

	\item The verifier samples $n_V \sample \{0,1\}^{\secparam}$ and sends
$(n_H, n_V)$ to the card.

	\item The card samples $n_C \sample \{0,1\}^{\secparam}$, computes $(D,
\pi_C) \sample \algo{Prove}_C(\key, \msg_1, n_V, n_C \| n_H)$, and sends $(n_C,
D, \pi_C)$ to the verifier.

	\item The verifier sends $(n_C, n_V, (v_i)_{i \in V}, D)$ to the holder.

	\item The holder computes $\Phi_H \sample \algo{Prove}_H(\key, Q,
(\msg_i)_{i=2}^{n}, \varphi, (v_i)_{i \in V}, n_C \| n_H, D, n_V)$ and sends it
to the verifier.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \algo{Verify}(\pk, (v_i)_{i \in V}, n_V, D,
\pi_C, \Phi_H)$ :] Return $1$ if and only if $\Phi_H = (E', \bar{E}, d, \pi_H)
\neq \perp$, the proofs $\pi_C$ and $\pi_H$ verify, and the following holds:
      %
      \begin{equation*}
      %
        \pairing(\bar{E}, \ggen) = \pairing(E', \ggen_Z).
      %
      \end{equation*}

  \end{description}

\end{primitivebox}

The card proves knowledge of the opening of $D$, a commitment to its message
under randomness $t$ that both provers derive from their shared PRF key. The
holder proves knowledge of the rest of the signature with $D$ in place of
$\gen_1^{\msg_1}$, using the proof of knowledge of a BBS+ signature of
Camenisch, Drijvers and Lehmann~\cite{EPRINT:CamDriLeh16}.

The joint proof used in $\OP$ extends their proof with what $\OP$ needs beyond
a proof of knowledge of a BBS+ signature. It differs from
\Cref{prim:hss-joint-proof} in five ways.

First, the secure element plays the card and the mobile device the holder. The
card's message is the user's secret key $\sk$, under $\gen_2$, and the
signature's randomness $r$ is under $\gen_3$.

Second, the secure element and the device communicate directly, since
registration pairs them. The secure element samples $t$ afresh and sends it to the
device, in place of both deriving it from a shared PRF key, and the payee takes
no part in producing the proof.

\begin{outline}
  \item $t$ is disclosed, so the device learns $\gen_2^{\sk}$; $\sk$ protected only by DL hardness, which is what the JPOK definition asks for
  \item difference from the original: there $t$ is derived from a PRF key shared between the provers so they never communicate (and $D$ is rerandomised per presentation); ours are paired at registration and talk directly, so the secure element samples $t$ fresh and sends it (old text had this in more detail)
\end{outline}

All the device receives from the secure element is the commitment
$D = \gen_2^{\sk}\gen_3^{t}$, its randomness $t$, and a proof of knowledge of the
opening, so the more powerful prover never sees $\sk$. This gives the
first-prover privacy required of a JPOK.

Third, the device shows the credential with the rerandomisation of Tessaro and
Zhu in place of the proof of Camenisch, Drijvers and
Lehmann~\cite{EC:TesZhu23b,EPRINT:CamDriLeh16}.

\begin{outline}
  \item a BBS+ signature cannot be shown directly: verification involves the signed messages, one of which is the identifier
  \item Tessaro-Zhu rerandomisation \cite{EC:TesZhu23b}: $E$ with two fresh exponents becomes $u, w, x$
  \item verification becomes one pairing equation on public values, $\pairing(u, \ggen_Z)\pairing(x, \ggen) = 1$, plus two discrete-logarithm equations proven by a Sigma protocol
  \item the check $u \neq 1$ rules out the trivial solution of the pairing equation
\end{outline}

Fourth, the proof is non-interactive and signs a message $m$. Both challenges
hash $m$ in place of the nonces of the verifier, the card and the holder.

Fifth, the device proves two further statements under its challenge, sharing
its response for $\algo{uid}$ with the credential equations: that the payer's
pseudonym $\com$ commits to $\algo{uid}$, and that the audit ciphertext $\ciph$
encrypts $\gen^{\algo{uid}}$.

\begin{outline}
  \item the payer's pseudonym: its response for $\algo{uid}$ is shared with the credential equation; this shared response ties pseudonym to credential (not needing this would be cool future work)\rebekah{you wrote ``not opening! just say payee and payer use one another's uids''; not sure which wording you want here}
  \item audit ciphertext must encrypt the identifier in the pseudonym; proving it alongside with the same $\algo{uid}$ response opens the pseudonym once per hop instead of twice
  \item the secure element's challenge hashes the signed message, so it must see the payment's destination
\end{outline}

\Cref{prot:bbs-joint-proof} gives the joint proof used in $\OP$.

\begin{protocolbox}{The joint proof used in $\OP$}{bbs-joint-proof}

  The public keys and the statement are as follows, with pseudonym $\com$,
audit ciphertext $\ciph = (c_1, c_2)$ and signed message $m$:
  %
  \begin{gather*}
  %
    \pk_\BBS = (\gen_0, \gen_1, \gen_2, \gen_3, \ggen, \ggen_Z), \quad \ck =
(\hgen_1, \hgen), \quad \pk_\auditor = \gen^{\mu_\auditor}, \\
  %
    \inst = (\pk_\BBS, \ck, \pk_\auditor, \com, \ciph, m).
  %
  \end{gather*}

  \begin{description}

    \item[$\Phi \sample \JPOKprove_0(\inst, \sk) \leftrightarrow
\JPOKprove_1(\inst, (\algo{uid}, \varphi, s, \rho))$ :] The secure element runs
$\JPOKprove_0$ and holds $\sk$. The device runs $\JPOKprove_1$ and holds the
credential
$\varphi = (e, r, E)$ on $(\algo{uid}, \sk)$, the opening $(\algo{uid}, s)$ of
$\com = \hgen_1^{\algo{uid}} \hgen^{s}$, and the randomness $\rho$ of $\ciph =
(\gen^{\rho}, \gen^{\algo{uid}} \pk_\auditor^{\rho})$.

      \begin{enumerate}

	\item The secure element samples $t, \gamma_2, \upsilon \sample \Zp$ and
computes the following:
          %
          \begin{gather*}
          %
            D = \gen_2^{\sk} \gen_3^{t}, \quad a_0 = \gen_2^{\gamma_2}
\gen_3^{\upsilon}, \quad \algo{chal}_0 = \hash_p(\gen_2, \gen_3, D, a_0, m), \\
          %
            \pi_2 = \gamma_2 + \algo{chal}_0\, \sk, \quad \zeta = \upsilon +
\algo{chal}_0\, t.
          %
          \end{gather*}
          %
          It sends $(D, t, a_0, \pi_2, \zeta)$ to the device.

	\item The device samples $\iota, \lambda \sample \Zp^*$, sets $r' = r
- t$, and computes the following:
          %
          \begin{equation*}
          %
            u = E^{\iota/\lambda}, \quad w = (\gen_0 \gen_1^{\algo{uid}} D
\gen_3^{r'})^{-1/\lambda}, \quad x = u^{e} w^{\iota}.
          %
          \end{equation*}

	\item The device samples $\alpha, \beta, \delta, \gamma_1, \gamma_3,
\gamma_s, \gamma_\rho \sample \Zp$ and computes the following:
          %
          \begin{gather*}
          %
            a = u^{\alpha} w^{\beta}, \quad b = w^{\delta} \gen_1^{\gamma_1}
\gen_3^{\gamma_3}, \quad d = \hgen_1^{\gamma_1} \hgen^{\gamma_s}, \quad a_1 =
\gen^{\gamma_\rho}, \quad a_2 = \gen^{\gamma_1} \pk_\auditor^{\gamma_\rho}, \\
          %
            \algo{chal}_1 = \hash_p(\inst, u, w, x, D, a_0, a, b, d, a_1, a_2),
\\
          %
            \eta = \alpha + \algo{chal}_1 e, \quad \nu = \beta + \algo{chal}_1
\iota, \quad \xi = \delta + \algo{chal}_1 \lambda, \\
          %
            \pi_1 = \gamma_1 + \algo{chal}_1 \algo{uid}, \quad \pi_3 =
\gamma_3 + \algo{chal}_1 r', \\
          %
            \pi_s = \gamma_s + \algo{chal}_1 s, \quad
\pi_\rho = \gamma_\rho + \algo{chal}_1 \rho.
          %
          \end{gather*}

	\item The device outputs $\Phi = (u,\allowbreak w,\allowbreak x,\allowbreak D,\allowbreak a_0,\allowbreak \pi_2,\allowbreak \zeta,\allowbreak a,\allowbreak b,
d,\allowbreak a_1,\allowbreak a_2,\allowbreak \eta,\allowbreak \nu,\allowbreak \xi,\allowbreak \pi_1,\allowbreak \pi_3,\allowbreak \pi_s,\allowbreak \pi_\rho)$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \JPOKverify(\inst, \Phi)$ :] Recompute
$\algo{chal}_0$ and $\algo{chal}_1$ as above, and return $1$ if and only if the
following hold:
      %
      \begin{gather*}
      %
        u \neq 1, \quad \pairing(u, \ggen_Z) \cdot \pairing(x, \ggen) = 1, \quad
\gen_2^{\pi_2} \cdot \gen_3^{\zeta} = a_0 \cdot D^{\algo{chal}_0}, \\
      %
        u^{\eta} \cdot w^{\nu} = a \cdot x^{\algo{chal}_1}, \quad w^{\xi} \cdot
\gen_1^{\pi_1} \cdot \gen_3^{\pi_3} = b \cdot (D \cdot
\gen_0)^{-\algo{chal}_1}, \quad \hgen_1^{\pi_1} \cdot \hgen^{\pi_s} = d \cdot
\com^{\algo{chal}_1}, \\
      %
        \gen^{\pi_\rho} = a_1 \cdot c_1^{\algo{chal}_1}, \quad \gen^{\pi_1}
\cdot \pk_\auditor^{\pi_\rho} = a_2 \cdot c_2^{\algo{chal}_1}.
      %
      \end{gather*}

  \end{description}

\end{protocolbox}

In $\OP$, the signed message is $m = (v, \com_i, \com_{i+1})$ for a payment and
$m = (v, \com_n, \algo{dep})$ for a deposit. The audit proof $\Gamma_i$ of the

\subsection{Instantiating $\SCI$ with Pointcheval-Sanders}\label{sec:offline-token-issuance}

The central bank's SCI scheme combines the Pointcheval-Sanders signature
scheme, which we abbreviate as PS, with the signing on committed inputs of the
Coconut credential scheme. \Cref{prim:ps} gives the signature scheme in the
variant whose signatures carry an extra random field element $\omega$. We index
its messages from $0$.

\begin{primitivebox}{The Pointcheval-Sanders signature scheme~\cite{RSA:PoiSan16,RSA:PoiSan18}}{ps}

  \begin{description}

    \item[$(\sk, \pk) \sample \PSgen(\secparam, \ell)$ :] Sample $\chi,
\theta_0, \ldots, \theta_\ell \sample \Zp^*$, and output $\sk = (\chi,
\theta_0, \ldots, \theta_\ell)$ and $\pk = (\tilde{x}, \tilde{y}_0, \ldots,
\tilde{y}_{\ell-1}, \tilde{z})$ with the following:
          %
          \begin{equation*}
          %
            \tilde{x} = \ggen^{\chi}, \quad \tilde{y}_i = \ggen^{\theta_i}, \quad
\tilde{z} = \ggen^{\theta_\ell}.
          %
          \end{equation*}

    \item[$\psi \sample \PSsign(\sk, \msg)$ :] On input $\msg \in
\Zp^{\ell}$, sample $k \sample \group{1} \setminus \{1\}$ and $\omega \sample
\Zp$, and output $\psi = (\omega, k, l)$ with $l$ as follows:
          %
          \begin{equation*}
          %
            l = k^{\chi + \sum_{i \in [\ell]_0} \theta_i \msg_i + \theta_\ell
\omega}.
          %
          \end{equation*}

    \item[$\{0,1\} \leftarrow \PSverify(\pk, \msg, \psi)$ :] On input $\psi =
(\omega, k, l)$, return $1$ if and only if the following hold:
          %
          \begin{equation*}
          %
            k \neq 1, \quad \pairing\Big(k, \tilde{x} \cdot \prod_{i \in [\ell]_0}
\tilde{y}_i^{\msg_i} \cdot \tilde{z}^{\omega}\Big) = \pairing(l, \ggen).
          %
          \end{equation*}

  \end{description}

\end{primitivebox}

Pointcheval and Sanders prove this variant existentially unforgeable under the
$q$-MSDH assumption of \Cref{asm:qmsdh}~\cite{RSA:PoiSan18}.

\begin{assumptionbox}{$q$-Modified Strong Diffie-Hellman ($q$-MSDH)}{qmsdh}

  The $q$-MSDH assumption holds if, given $(\gen^{\chi^i},
\ggen^{\chi^i})_{i=0}^{q}$ and $(\gen^{\varsigma}, \ggen^{\varsigma},
\ggen^{\chi\varsigma})$ for uniformly random $\chi, \varsigma \sample \Zp^*$, no
PPT adversary can output a tuple $(c, \mathcal{P}, Q, Q^{1/(\chi+c)},
Q^{\varsigma/\mathcal{P}(\chi)})$ with non-negligible probability, where $Q \in
\group{1}$ is not the identity and $\mathcal{P}$ is a polynomial of degree at most
$q$ with $\mathcal{P}(-c) \neq 0$.

\end{assumptionbox}

Coconut signs committed inputs with the Pointcheval-Sanders signature scheme
without $\omega$. To request a signature, the recipient commits to the messages,
encrypts each message in the exponent of $k = \hash_{\group{1}}(C)$ under a
fresh ElGamal key of its own, and proves that the commitment and the
ciphertexts are consistent. The signer signs homomorphically on the
ciphertexts, and the recipient decrypts. \Cref{prim:coconut} gives the protocol
for a single signer.

\begin{primitivebox}{Signing on committed inputs in Coconut~\cite{NDSS:SABMD19}}{coconut}

  The signer's key pair is $\sk = (\chi, \theta_1, \ldots, \theta_\ell)$ and
$\pk = (\tilde{x}, \tilde{y}_1, \ldots, \tilde{y}_\ell)$, with $\tilde{x} =
\ggen^{\chi}$ and $\tilde{y}_j = \ggen^{\theta_j}$, and a signature on $\msg =
(\msg_1, \ldots, \msg_\ell)$ is $(k, k^{\chi + \sum_{j=1}^{\ell} \theta_j
\msg_j})$. The generators $\gen, P_1, \ldots, P_\ell$ of $\group{1}$ are public.

  \begin{description}

    \item[$(\mu, \Lambda) \sample \algo{PrepareBlindSign}(\msg)$ :] Sample
$\mu, \openinfo, \rho_1, \ldots, \rho_\ell \sample \Zp$, set $B = \gen^{\mu}$,
and compute the following:
          %
          \begin{equation*}
          %
            C = \gen^{\openinfo} \prod_{j=1}^{\ell} P_j^{\msg_j}, \quad k =
\hash_{\group{1}}(C), \quad \hat{\ciph}_j = (\gen^{\rho_j}, B^{\rho_j}
k^{\msg_j}) \ \text{for } j \in [\ell].
          %
          \end{equation*}
          %
Compute the following proof:
          %
          \begin{multline*}
          %
            \pi_s \sample \NIZK\bigl\{(\mu, \msg, \openinfo, \rho_1, \ldots,
\rho_\ell) : B = \gen^{\mu} \ \wedge\ C = \gen^{\openinfo} \prod_{j=1}^{\ell}
P_j^{\msg_j} \\
          %
            \wedge\ \hat{\ciph}_j = (\gen^{\rho_j}, B^{\rho_j} k^{\msg_j}) \
\text{for } j \in [\ell]\bigr\}.
          %
          \end{multline*}
          %
Output $\mu$ and $\Lambda = (B, C, (\hat{\ciph}_j)_{j \in [\ell]}, \pi_s)$.

    \item[$\tilde{\psi} \leftarrow \algo{BlindSign}(\sk, \Lambda)$ :]
Recompute $k = \hash_{\group{1}}(C)$, and output $\perp$ if $\pi_s$ does not
verify. Otherwise parse $\hat{\ciph}_j = (f_{j,1}, f_{j,2})$, and output
$\tilde{\psi} = (k, f_1, f_2)$ with $f_1$ and $f_2$ as follows:
          %
          \begin{equation*}
          %
            f_1 = \prod_{j=1}^{\ell} f_{j,1}^{\theta_j}, \quad f_2 = k^{\chi}
\prod_{j=1}^{\ell} f_{j,2}^{\theta_j}.
          %
          \end{equation*}

    \item[$\psi \leftarrow \algo{Unblind}(\tilde{\psi}, \mu)$ :] Parse
$\tilde{\psi} = (k, f_1, f_2)$, and output $\psi = (k, f_2 / f_1^{\mu})$.

  \end{description}

\end{primitivebox}

Coconut also lets several signers each issue a share of the signature, and lets
$\pi_s$ prove an application-specific statement about the messages. We use
neither. Coconut compiles its proofs with the Fiat-Shamir transform.

\Cref{prim:coconut} with the following changes. It signs with the variant of

them to the exponent of $k$ directly. In $\OP$ the disclosed message
is the token value. The commitment randomness $\openinfo$ is under its own
generator $P_\ell$ rather than $\gen$, and the recipient checks the unblinded
signature with $\PSverify$. The proof $\algoproof_{\algo{req}}$ does not show

\begin{protocolbox}{Signing on committed inputs in $\OP$}{offline-sci}

  \begin{description}

    \item[$(\sk, \pk) \sample \SCIgen(\secparam, \ell)$ :] Compute $(\sk,
\pk_{\PS}) \sample \PSgen(\secparam, \ell)$, and output $\sk$ and $\pk =
(\pk_{\PS}, \vec{P})$, where $\vec{P} = (P_0, \ldots, P_\ell)$ are the
generators for committed inputs.

    \item[$(\algo{req}, \state) \sample \SCIrequest(\pk, \msg, \algo{Disc})$ :]
On input $\msg \in \Zp^{\ell}$ and a set $\algo{Disc} \subseteq [\ell]_0$ of
indices to disclose, let $H = [\ell]_0 \setminus \algo{Disc}$. Sample $\openinfo,
\mu \sample \Zp$, set $B = \gen^{\mu}$, and compute the following:
          %
          \begin{equation*}
          %
            C = P_\ell^{\openinfo} \prod_{i \in H} P_i^{\msg_i}, \quad k =
\hash_{\group{1}}(C), \quad \hat{\ciph}_i = (\gen^{\rho_i}, k^{\msg_i}
B^{\rho_i}) \ \text{for } i \in H,
          %
          \end{equation*}
          %
with $\rho_i \sample \Zp$. Compute $\algoproof_{\algo{req}}$, a non-interactive
proof that $C = P_\ell^{\openinfo} \prod_{i \in H} P_i^{\msg_i}$ and
$\hat{\ciph}_i = (\gen^{\rho_i}, k^{\msg_i} B^{\rho_i})$ for every $i \in H$, on
witness $(\openinfo, (\msg_i, \rho_i)_{i \in H})$ and statement $(\algo{Disc},
\vec{P}, \gen, B, C, k, (\hat{\ciph}_i)_{i \in H})$. Output $\algo{req} = (C,
B, (\msg_i)_{i \in \algo{Disc}}, (\hat{\ciph}_i)_{i \in H},
\algoproof_{\algo{req}})$ and $\state = (\pk, \msg, \mu, k)$.

    \item[$\algo{resp} \sample \SCIissue(\sk, \algo{req})$ :] Recompute $k =
\hash_{\group{1}}(C)$, and output $\perp$ if $\algoproof_{\algo{req}}$ does not
verify. Otherwise sample $\omega \sample \Zp$, parse $\hat{\ciph}_i = (f_{i,1},
f_{i,2})$, and output $\algo{resp} = (\omega, f_1, f_2)$ with $f_1$ and $f_2$
as follows:
          %
          \begin{equation*}
          %
            f_1 = \prod_{i \in H} f_{i,1}^{\theta_i}, \quad f_2 = k^{\chi +
\theta_\ell \omega + \sum_{i \in \algo{Disc}} \theta_i \msg_i} \prod_{i \in H}
f_{i,2}^{\theta_i}.
          %
          \end{equation*}

    \item[$\psi \leftarrow \SCIunblind(\state, \algo{resp})$ :] Compute $l = f_2
/ f_1^{\mu}$, and output $\psi = (\omega, k, l)$ if $\SCIverify(\pk, \msg, \psi)
= 1$, and $\perp$ otherwise.

    \item[$\{0,1\} \leftarrow \SCIverify(\pk, \msg, \psi)$ :] Return
$\PSverify(\pk_{\PS}, \msg, \psi)$.

  \end{description}

\end{protocolbox}

  \item one-more unforgeability: $q$-MSDH plus the analysis of Coconut \cite{NDSS:SABMD19}

  \item online request extractability: \rebekah{Fischlin: argued from a non-rewinding extractor for $\algoproof_{\algo{req}}$ given by the Fischlin transform; only if confirmed necessary, the code uses plain Fiat-Shamir}

In the withdrawal step of \Cref{prot:offline}, the device runs $\SCIrequest$ on
the messages $(\algo{uid}_0, v, s_0)$ with $\algo{Disc} = \{1\}$, so the central
bank sees only the value. It sets $\openinfo = 0$ rather than sampling it, since
the hidden message $s_0$ is uniform and already hides $C$.\rebekah{$\SCIrequest$ as written samples $\openinfo$; either make $\openinfo$ an input or say in \Cref{prot:offline} that the withdrawal fixes it to $0$}

The device does not show $\psi$ to payees, for the reason given in
\Cref{sec:offline-construction}. It instead proves once that it holds a valid signature
on the token value and on the identifier and randomness committed in its
pseudonym. It rerandomises $(k, l)$ to $(k^*, l^*)$ and proves the remaining
equation by a Sigma protocol whose challenge hashes the value, the pseudonym and
$(k^*, l^*)$. The proof of withdrawal $\Sigma$ is the first entry of every
token's history, and every payee verifies it.

\begin{outline}
  \item why the challenge hashes value, pseudonym and $(k^*, l^*)$: none of them can be swapped later
\end{outline}

Since $s_0$ is random and not known, $(k^*, l^*)$ leaks no information about
$\algo{uid}_0$.\rebekah{$(k^*, l^*, \tilde{c})$ are part of the statement, not hidden by zero knowledge, so this needs its own argument; nothing in the proof shows that the central bank cannot link $\Sigma$ to the withdrawal it signed} \Cref{prot:ps-proof} gives the proof of withdrawal.

\begin{protocolbox}{The proof of withdrawal}{ps-proof}

  Let $\pk_{\algo{PS}} = (\tilde{x}, \tilde{y}_0, \tilde{y}_1, \tilde{y}_2,
\tilde{z}, \vec{P})$ be the central bank's public key, and $\tau = (v, \com)$ a
token with $\com = \hgen_1^{\algo{uid}} \hgen^{s}$ and $\psi = (\omega, k, l)$ a
Pointcheval-Sanders signature on $(\algo{uid}, v, s)$.

  \begin{description}

    \item[$\Sigma \sample \algo{ProveWithdrawal}(\pk_{\algo{PS}}, \tau,
\algo{uid}, s, \psi)$ :] Do the following:

      \begin{enumerate}

	\item Sample $\vartheta \sample \Zp^*$ and set $(k^*, l^*) =
(k^{\vartheta}, l^{\vartheta})$.

	\item Compute $\tilde{c} = \tilde{y}_0^{\algo{uid}} \tilde{y}_2^{s}
\tilde{z}^{\omega}$.

	\item Sample $\alpha, \beta, \delta \sample \Zp$ and compute the
following:
          %
          \begin{gather*}
          %
            a = \hgen_1^{\alpha} \hgen^{\beta}, \quad \tilde{a} =
\tilde{y}_0^{\alpha} \tilde{y}_2^{\beta} \tilde{z}^{\delta}, \\
          %
            \algo{chal} = \hash_p(\pk_{\algo{PS}}, \ck, v, \com, k^*, l^*,
\tilde{c}, a, \tilde{a}).
          %
          \end{gather*}

	\item Compute the following:
          %
          \begin{equation*}
          %
            \pi_{\algo{uid}} = \alpha + \algo{chal}\, \algo{uid}, \quad \pi_s =
\beta + \algo{chal}\, s, \quad \pi_\omega = \delta + \algo{chal}\, \omega.
          %
          \end{equation*}

	\item Output $\Sigma = (k^*, l^*, \tilde{c}, a, \tilde{a},
\pi_{\algo{uid}}, \pi_s, \pi_\omega)$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \algo{VerifyWithdrawal}(\pk_{\algo{PS}}, \tau,
\Sigma)$ :] Recompute $\algo{chal}$ as above, and return $1$ if and only if the
following hold:
      %
      \begin{gather*}
      %
        k^* \neq 1, \quad \pairing(l^*, \ggen) = \pairing(k^*, \tilde{x} \cdot
\tilde{c} \cdot \tilde{y}_1^{v}), \\
      %
        \hgen_1^{\pi_{\algo{uid}}} \cdot \hgen^{\pi_s} = a \cdot
\com^{\algo{chal}}, \quad \tilde{y}_0^{\pi_{\algo{uid}}} \cdot
\tilde{y}_2^{\pi_s} \cdot \tilde{z}^{\pi_\omega} = \tilde{a} \cdot
\tilde{c}^{\algo{chal}}.
      %
      \end{gather*}

  \end{description}

\end{protocolbox}
