\section{Definitions}
% %
% % We formalise the security of anonymous aggregate credentials following the
% % ideal world paradigm, using a simulation based definition.  In the ideal world,
% % we find a functionality $\F{\ATS}$, which by construction, captures the desired
% % security and privacy properties of anonymous aggregate credentials, together
% % with a simulator $\simu$ that controls a subset of issuers, users, and
% % verifiers.  On the other hand, in the real world, we find a candidate protocol
% % $\Pi$ and an adversary $\adv$ that controls the same subset of system
% % participants as $\simu$, and can instruct them to behave arbitrarily.  We use
% % the real ideal world paradigm to capture all interactions between issuers,
% % users, and verifiers in one instance. We consider a static corruption model
% % that allows the simulator $\simu$ (and adversary $\adv$) to corrupt any number
% % of issuers, users and verifiers but before any calls to $\F{\ATS}$'s interfaces
% % (or $\Pi$'s functions) take place.  We note that since the registration
% % authority is honest but curious, $\simu$ will not corrupt it. It will be given
% % access to any information that the registration authority learns during its
% % interactions with various system participants.
% %
% % \paragraph{Ideal Functionality $\F{\ATS}$.} The ideal functionality represents
% % a trusted third party, which executes the operations of aggregate anonymous
% % credentials on behalf of system participants in a way that perfectly meets the
% % system requirements.  $\F{\ATS}$ receives inputs from the system participants
% % and simulator $\simu$ and correctly generate the corresponding outputs, which
% % are then transmitted to the parties authorised to access them.  $\F{\ATS}$
% % exposes five interfaces: register, issue, verify. $\simu$ and honest system
% % participants submit their inputs through these interfaces.
% %
% % At a high level, the ideal functionality captures the properties of aggregate
% % anonymous credentials: unforgeability, unlinkability, and aggregatability.
% % Unforgeability is enforced by ensuring that only validly issued attributes can
% % be successfully presented or verified. Unlinkability is achieved as randomness
% % is returned in each query to present, which prevents presentations from being
% % correlated. Aggregation occurs through calling the $\ATSaggregate$ interface.
% %
% % \paragraph{Register.} This is called by a user to register into the system.  If
% % the registration authority accepts, and the user has never been registered
% % before, the user is successfully registered.
% %
% % \paragraph{Issuance.} This is invoked when a user wants to get a credential
% % that certifies value $\atv{}$ of some attribute $\att{}$.  If the authorised
% % issuer accepts the issuance request, the user obtains a credential that they
% % can use in subsequent presentations.
% %
% % During a call, $\simu$ learns the identity of the user. During a call to issue,
% % $\simu$ additionally learns the value of the user's attribute.
% %
% % \paragraph{Aggregation.} This is invoked by a user to aggregate the user's
% % credentials for attribute values $\vec{\atv{}}$ in a way that enables them to
% % present the aggregate credential later.  Aggregation is a local operation, and
% % so $\simu$ does not learn any information during the call.  A successful call
% % results in the user receiving a random string $\signature$ representing the
% % aggregate credentials.
% %
% % \paragraph{Presentation.} Present is queried by a user to produce a
% % presentation of attribute values ($\vec{\atv{}}$).  the user 's request
% % consists of triple $(\vec{\atv{}}, \signature, \rand)$, with $\signature$ the
% % aggregation of the credentials certifying $\vec{\atv{}}$ and $\rand$ a random
% % number that identifies this presentation request.  If the user has previously
% % successfully called aggregate with $\vec{\atv{}}$, then $\F{\ATS}$ accepts the
% % query. Otherwise, it checks if the user actually holds the $\vec{\atv{}}$.
% % This captures the fact that even if a corrupt the user does not call aggregate
% % before calling present, $\F{\ATS}$ may still accept the presentation request as
% % long as the user has previously received the relevant credentials.  If
% % $\F{\ATS}$ accepts, then it generates a random string $\proof$; the randomness
% % of $\proof$ ensures that triple $(\vec{\atv{}}, \msg, \proof)$ discloses no
% % information about the user  beyond what's revealed by $\vec{\atv{}}$.
% %
% % \paragraph{Verification.} Verify is invoked by a verifier to check the validity
% % of credential presentation $(\vec{\atv{}}, \msg, \proof)$.  If the presentation
% % has already been produced by a valid present call, then $\F{\ATS}$ accepts and
% % returns true.  If it has already been flagged as invalid, then $\F{\ATS}$
% % rejects and returns false.  Finally, if the presentation has never been
% % received before, then $\F{\ATS}$ verifies whether the presentation could have
% % been created by corrupt users and issuers.  That is, $\F{\ATS}$ checks if there
% % are corrupt users that hold all the disclosed attribute values, and if not,
% % then $\F{\ATS}$ also checks if there are corrupt issuers that can produce valid
% % credentials for the attribute values not held by corrupt users.  In these
% % checks succeed, then $\F{\ATS}$ refers to $\simu$ and returns what $\simu$
% % returns; otherwise, it records the presentation, flags it as invalid, and
% % outputs false.
% %

% % Let  $\textsc{Exec}_{\Pi,\adv}(\secparam)$ denote the outputs of the honest
% % parties and~$\adv$, during an execution of~$\Pi$ in the real world and

% % \begin{definition} A protocol~$\Pi$ securely realises} ideal

% % \textsc{Exec}_{\F{\ATS},\simu}(\secparam)$, where $\secparam$ is the security
% % parameter, and $\approx_c$ denotes computational indistinguishability.

% % \subsection{Aggregate Tag-Based Signatures (ATS)}
% %
% % We define an aggregate tag-based signature scheme as one that allows signatures
% % computed over the same tag to be aggregated into a compact signature. An
% % aggregate tag-based signature scheme is defined by the following algorithms:
% %
% % \begin{itemize}
% %
% %   \item $(\sk, \pk) \leftarrow \ATSgen(\secparam)$: on input security parameter
% %     $\secparam$, output key pair $(\sk,\pk)$.
% %
% %   \item $\signature \leftarrow \ATSsign(\sk, \tau, \msg)$: On input secret key
% %     $\sk$, tag $\tau \in \{0,1\}^*$, and message $\msg \in \{0,1\}^*$, output
% %     signature $\signature$.
% %
% %   \item $ \signature' \leftarrow \ATSrandomise(\tau, \signature)$: On input tag
% %     $\tau$ and signature $\signature$, output randomised signature
% %     $\signature'$.
% %
% %   \item $ \{0,1\} \leftarrow \ATSverify(\pk, \tau, \msg, \signature)$: on input
% %     public key $\pk$, tag $\tau$, message $\msg$, and signature $\signature$,
% %     return 0 or 1.
% %
% % \item $\signature \leftarrow \ATSaggregate(\signature_1, \ldots,
% % \signature_n)$: on input signatures $\signature_1,\ldots,\signature_n$ (all
% % under the same} tag $\tau$), output aggregate signature $\signature$.
% %
% % \item $\{0,1\} \leftarrow \ATSverifyagg((\pk_i)_{i\in [1 \isep n]}, \tau,
% %   (\msg_i)_{i\in [1 \isep n]}, \signature)$: on input public keys
% %   $(\pk_i)_{i\in [1 \isep n]}$, tag $\tau$, messages $(\msg_i)_{i\in [1 \isep
% %   n]}$, and aggregate signature $\signature$, return 0 or 1.
% %
% % \end{itemize}
% %
% % An aggregate tag based signature scheme should satisfy correctness and
% % existentially unforgeability, as defined below.
% %
% % \begin{definition}[Correctness] An ATS scheme satisfies correctness if for all
% %   security parameters $\secparam$, all $n \geq 1$, all tags $\tau \in
% %   \{0,1\}^*$, and all messages $\msg_i \in \{0,1\}^*$, the following holds.
% %
% %   Let $\ind \subseteq [1 \isep n]$. For all $i \in \ind$, sample $(\sk_i,
% %   \pk_i) \gets \ATSgen(\secparam)$ and let $\signature_i \gets \ATSsign(\sk_i,
% %   \tau, \msg_i)$.  Then $\signature$ is produced as follows: $$ \signature
% %   \gets \ATSaggregate((\signature_i)_{i \in \ind}).$$
% %
% %   Then we have the following: $$ \forall i \in \ind : \ATSverify(\pk_i, \tau,
% %   \msg_i, \signature_i) = 1 \ \ \land \ \ \ATSverifyagg\bigl((\pk_i)_{i \in
% %   \ind}, \tau, (\msg_i)_{i \in \ind}, \signature\bigr) = 1.$$
% %
% % Randomisation also preserves validity, meaning that for all $\signature'
% % \leftarrow \ATSrandomise(\tau, \signature)$ and $\ATSverify(\pk, \tau, \msg,
% % \signature) = 1$, we also have $\ATSverify(\pk, \tau, \msg, \signature') = 1$
% % \end{definition}
% %

% %   corruption oracle $\Corrupt$ and signing oracle $\Sign$, the probability that
% % $\adv$ wins the experiment depicted in \fref{fig:exp} is negligible in security
% % parameter $\secparam$.  \end{definition}
