\section{Instantiation}\label{sec:tsps-instantiation}

This section instantiates the construction of \Cref{sec:tsps-construction}
twice, giving full descriptions of two threshold structure preserving signature
schemes built from the MPC protocol of
\Cref{sec:tsps-mpc}~\cite{C:AGHO11,C:KilPanWee15}. We first give the two
underlying structure preserving signature schemes, then verify in each case
that the signing algorithm admits the decomposition the construction requires,
and give the resulting threshold scheme. Both schemes are designed with a
non-interactive online phase in mind, with $\algo{OffSign}$ carrying
essentially all of the interactive computation.

\subsection{Instantiating $\SPS$}

In this section we describe two structure preserving signatures. The first is
optimal in terms of size, and existentially unforgeable in the generic group
model (GGM)~\cite{C:AGHO11}. The second produces signatures with more group
elements, but is existentially unforgeable under the DDH
assumption~\cite{C:KilPanWee15}.

\subsubsection{Optimal re-randomisable structure preserving signatures in the
GGM}\label{sec:tsps-sps-ggm}

Abe, Groth, Haralambiev, and Ohkubo introduced the first optimal-size
structure preserving signature scheme for asymmetric pairings~\cite{C:AGHO11}.

\begin{primitivebox}{Optimal structure preserving signatures (Abe, Groth,
  Haralambiev, and Ohkubo)}{sps-ggm}

  \begin{description}

    \item[$(\pk, \sk) \sample \SPSgen(\secparam, \ell)$ :] On input security
      parameter $\secparam$ and vector length $\ell$, do:

      \begin{enumerate}

        \item Sample $\vec{\mu} = (\mu_1, \ldots, \mu_\ell) \sample
          (\Zp^*)^\ell$ and $\tau \sample \Zp$.

        \item Compute $\vec{\widetilde{u}} = (\widetilde{u}_1, \ldots,
          \widetilde{u}_\ell) = (\mu_1 \ggen, \ldots, \mu_\ell \ggen) \in
          \group{2}^\ell$ and $v = \tau \gen \in \group{1}$.

        \item Return $\pk = (\vec{\widetilde{u}}, v)$ and $\sk = (\vec{\mu},
          \tau)$.

      \end{enumerate}

    \item[$\SPSsig \sample \SPSsign(\sk, \vec{\msg})$ :] On input secret key $\sk
      = (\vec{\mu}, \tau)$ and vector $\vec{\msg} = (\msg_1, \ldots, \msg_\ell) \in
      \group{1}^\ell$, do:

      \begin{enumerate}

        \item Sample $\rho \sample \Zp^*$.

        \item Return $\SPSsig = (\widetilde{r}, \widetilde{s}, z)$ such
          that the following holds:
          %
          \begin{equation*}
          %
            \widetilde{r} = \tfrac{1}\rho\ggen, \qquad \widetilde{s} =
            \tau\widetilde{r}, \qquad z = \rho\left(\gen -
            \vec{\mu}\vec{\msg}\right).
          %
          \end{equation*}

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \SPSverify(\pk, \vec{\msg}, \SPSsig)$ :] On input
      signature $\SPSsig = (\widetilde{r}, \widetilde{s}, z) \in
      \group{2}^2 \times \group{1}$ and vector $\vec{\msg} \in \group{1}^\ell$,
      return 1 if the following holds, and 0 otherwise:
      %
      \begin{equation}\label{eq:abe-ver}
      %
        \pairing(v, \widetilde{r}) = \pairing(\gen, \widetilde{s}) \ \wedge \
        \pairing(z, \widetilde{r})\prod_{i=1}^\ell\pairing(\msg_i,
        \widetilde{u}_i) = \pairing(\gen, \ggen).
      %
      \end{equation}

  \end{description}

\end{primitivebox}

This signature is existentially unforgeable in the GGM~\cite{C:AGHO11}.

\subsubsection{Structure preserving signatures under
DDH}\label{sec:tsps-sps-kiltz}

Kiltz, Pan, and Wee introduced a structure preserving signature scheme that is
existentially unforgeable under the DDH assumption~\cite{C:KilPanWee15}.

\begin{primitivebox}{Structure preserving signatures under DDH (Kiltz, Pan, and
  Wee)}{sps-kiltz}

  \begin{description}

    \item[$(\pk, \sk) \sample \SPSgen(\secparam, \ell)$ :] On input security
      parameter $\secparam$ and vector length $\ell$, do:

      \begin{enumerate}

        \item Sample column vectors $\vec{\alpha} = (1, \alpha)^{\mathsf{T}}$
          and $\vec{\beta} = (1, \beta)^{\mathsf{T}}$, for uniformly random
          $\alpha, \beta \in \Zp$, together with uniformly random matrices
          $\mat{K} \in \Zp^{(\ell+1)\times 2}$,
          $\mat{\Gamma} \in \Zp^{2\times 2}$, and $\mat{\Delta}
          \in \Zp^{2\times 2}$.

        \item Compute $\vec{\mu} = \mat{K}\vec{\alpha}$, $\vec{\nu}
          = \mat{\Gamma}\vec{\alpha}$, $\vec{\omega} =
          \mat{\Delta}\vec{\alpha}$, $\vec{\xi} =
          \vec{\beta}^{\mathsf{T}}\mat{\Gamma}$, and $\vec{\upsilon} =
          \vec{\beta}^{\mathsf{T}}\mat{\Delta}$.

        \item Compute $\vec{b} = \gen(1, \beta)$, $\vec{x} =
          \gen\vec{\xi}$, and $\vec{y} = \gen\vec{\upsilon}$.

        \item Compute $\vec{\widetilde{a}} = \ggen(1, \alpha)$,
          $\vec{\widetilde{u}} = \ggen\vec{\mu}$, $\vec{\widetilde{v}} =
          \ggen\vec{\nu}$, and $\vec{\widetilde{w}} = \ggen\vec{\omega}$.

        \item Return $\sk = (\mat{K}, \vec{b}, \vec{x},
          \vec{y})$ and $\pk = (\vec{\widetilde{a}}, \vec{\widetilde{u}},
          \vec{\widetilde{v}}, \vec{\widetilde{w}})$.

      \end{enumerate}

    \item[$\SPSsig \sample \SPSsign(\sk, \vec{\msg})$ :] On input secret key $\sk
      = (\mat{K}, \vec{b}, \vec{x}, \vec{y})$ and vector
      $\vec{\msg} = (\msg_1, \ldots, \msg_\ell)$, do:

      \begin{enumerate}

        \item Sample $\rho, \sigma \sample \Zp^*$.

        \item Compute:
          %
          \begin{align*}
          %
            \widetilde{q} &= \sigma\ggen, \qquad \vec{r} = \rho\vec{b},
            \qquad \vec{s} = \rho\sigma\vec{b}, \\ \vec{z} &= (\gen,
            \vec{\msg})\mat{K} + \rho (\vec{x}+\sigma\vec{y}).
          %
          \end{align*}

        \item Return $\SPSsig = (\widetilde{q}, \vec{r}, \vec{s},
          \vec{z})$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \SPSverify(\pk, \vec{\msg}, \SPSsig)$ :] On input
      signature $\SPSsig =(\widetilde{q}, \vec{r}, \vec{s}, \vec{z}) \in
      \group{2}\times\group{1}^6$ and vector $\vec{\msg}$ of $\ell$ messages in
      $\group{1}$, return 1 if the following holds, and 0 otherwise:
      %
      \begin{equation}\label{eq:kiltz-ver}
      %
        \pairing(\vec{z}, \vec{\widetilde{a}}) = \pairing((\gen, \vec{\msg}),
        \vec{\widetilde{u}})\, \pairing(\vec{r},\vec{\widetilde{v}})\,
        \pairing(\vec{s}, \vec{\widetilde{w}}) \ \wedge \ \pairing(r,
        \widetilde{q}) = \pairing(s, \ggen).
      %
      \end{equation}

  \end{description}

\end{primitivebox}

\subsection{Instantiating $\TSPS$}

\subsubsection{Optimal threshold structure preserving
signatures}\label{sec:tsps-abe}

We formally describe a threshold structure preserving signature scheme based on
the structure preserving scheme of Abe, Groth, Haralambiev, and Ohkubo, built
from the MPC protocol of \Cref{sec:tsps-mpc}~\cite{C:AGHO11}. The scheme is
given in \Cref{prot:tsps-abe}. Its signing algorithm has the decomposition the
construction requires: the randomness $\rho$ and every value derived from it
are independent of the message, and the message enters only through terms of
the form $\mu_i\rho\msg_i$, in which a secret-shared field element multiplies a
public group element.

\begin{protocolbox}{Optimal threshold structure preserving
  signatures~\cite{C:AGHO11}}{tsps-abe}

  \begin{description}

    \item[$(\pk, \shr{\sk}) \sample \TSPSgen(\secparam, \ell, t, n)$ :] On
      input the security parameter, vector length $\ell$, corruption threshold
      $t$, and number of signers $n$, the honest signers do the following:

      \begin{enumerate}

        \item Issue $\ell+1$ parallel calls to $\algo{RandShare}$ to compute
          $t$-additive shares $(\shr{\mu_1}, \ldots, \shr{\mu_\ell},
          \shr{\tau})$.

        \item Each signer locally computes $\shr{v} = \shr{\tau}\gen$ and
          $\shr{\widetilde{u}_i} = \shr{\mu_i}\ggen$ for $i \in [\ell]$,
          multiplying its own share by the relevant generator.

        \item Query $\algo{Open}$ with the shares $\shr{v}$, and
          $\algo{Open}$ with the shares $(\shr{\widetilde{u}_1}, \ldots,
          \shr{\widetilde{u}_\ell})$.

        \item If no $\bot$ occurs, return public key $\pk =
          (\vec{\widetilde{u}}, v)$ with $\vec{\widetilde{u}} =
          (\widetilde{u}_1, \ldots, \widetilde{u}_\ell)$, with corresponding
          secret shares $\shr{\sk} = (\shr{\mu_1}, \ldots, \shr{\mu_\ell},
          \shr{\tau})$ distributed to the signers.

      \end{enumerate}

    \item[$\shr{\SPSsig}_{\algo{OffSign}} \sample \algo{OffSign}(\shr{\sk})$
      :] On input the secret key shares, the honest signers do the following:

      \begin{enumerate}

        \item Invoke $\algo{RandShare}$ to get $t$-additive shares
          $\shr{\rho}$.

        \item If no $\bot$ occurs, call $\algo{Inv}$ to compute $t$-additive
          shares $\shr{1/\rho}$.

        \item Issue, potentially in parallel, $\ell+1$ calls to $\algo{Mul}$ to
          compute the $t$-additive shares $(\shr{\rho\mu_1},
          \ldots,\shr{\rho\mu_\ell}, \shr{\tau/\rho})$.

        \item If the calls to $\algo{Inv}$ and $\algo{Mul}$ all succeed, the
          signers hold shares $(\shr{1/\rho}, \shr{\tau/\rho}, \shr{\rho\mu_1},
          \ldots, \shr{\rho\mu_\ell})$.

        \item Each signer locally computes $\shr{\rho\gen}$ from $\shr{\rho}$,
          and $\shr{\widetilde{r}} = \shr{\tfrac{1}\rho\ggen}$ and
          $\shr{\widetilde{s}} = \shr{\tfrac{\tau}\rho\ggen}$ from
          $\shr{1/\rho}$ and $\shr{\tau/\rho}$, against the relevant
          generators.

        \item Conclude the offline phase by storing $(\shr{\rho\mu_1} , \ldots,
          \shr{\rho\mu_\ell}, \shr{\rho\gen}, \shr{\widetilde{r}},
          \shr{\widetilde{s}})$.

      \end{enumerate}

    \item[$\shr{\SPSsig}_{\algo{OnSign}} \sample \algo{OnSign}(\vec{\msg})$ :] On
      input message vector $\vec{\msg}$, the honest signers do the following:

      \begin{enumerate}

        \item For each $\msg_i \neq 0$, each signer locally multiplies its
          share $\shr{\mu_i\rho}$ by the public group element $-\msg_i$ to
          obtain $t$-additive shares $\shr{-\mu_i\rho\msg_i}$.

        \item Each signer locally adds these to obtain its share $\shr{z} =
          \shr{\rho\gen + \sum_{i=1, \msg_i \neq 0}^\ell {-\mu_i\rho\msg_i}}$,
          equivalent to $\rho\gen - \sum_{i=1}^\ell \mu_i\rho\msg_i$. Both
          steps are local, so $\algo{OnSign}$ requires no interaction.

      \end{enumerate}

    \item[$\SPSsig \leftarrow \TSPScombine(\shr{\SPSsig})$ :] On input
      the signature shares, the honest signers do the following:

      \begin{enumerate}

        \item Call $\algo{Open}$ with $\shr{\widetilde{r}}$ and
          $\shr{\widetilde{s}}$ to obtain $\widetilde{r} =
          \tfrac{1}\rho\ggen$ and $\widetilde{s} = \tfrac{\tau}\rho\ggen$,
          and $\algo{Open}$ with shares $\shr{z}$ to compute $z = \rho(\gen
          - \sum_{i=1}^\ell \mu_i\msg_i)$.

        \item If no $\bot$ occurs, return the valid signature $\SPSsig =
          (\widetilde{r}, \widetilde{s}, z)$ on the vector $(\msg_1,
          \ldots,\msg_\ell) \in \group{1}^\ell$.

      \end{enumerate}

    \item[$\{0,1\} \leftarrow \SPSverify(\pk, \vec{\msg}, \SPSsig)$ :] On input
      signature $\SPSsig = (\widetilde{r}, \widetilde{s}, z)$, vector
      $(\msg_1, \ldots, \msg_\ell) \in \group{1}^\ell$, and public key $\pk =
      (\vec{\widetilde{u}}, v)$, do the following:

      \begin{enumerate}

        \item Check whether~\eqref{eq:abe-ver} holds, returning 1 if so and 0
          otherwise.

      \end{enumerate}

  \end{description}

\end{protocolbox}

\subsubsection{Threshold structure preserving signatures from
DDH~\cite{C:KilPanWee15}}\label{sec:tsps-kiltz}

We now formally describe a threshold structure preserving signature scheme
based on the structure preserving scheme of Kiltz, Pan, and Wee, built from the
MPC protocol of \Cref{sec:tsps-mpc}~\cite{C:KilPanWee15}. The scheme is given
in \Cref{prot:tsps-kiltz}. Its signing algorithm again has the required
decomposition: $\rho$, $\sigma$, and the terms derived from them are
message-independent, and the message enters only through $(\gen,
\vec{\msg})\mat{K}$, in which the secret-shared entries of
$\mat{K}$ multiply the public group elements of $\vec{\msg}$.

\begin{protocolbox}{Threshold structure preserving signature based
  on~\cite{C:KilPanWee15}}{tsps-kiltz}

  \begin{description}

    \item[$(\pk, \shr{\sk}) \sample \TSPSgen(\secparam, \ell, t, n)$ :] On
      input the security parameter, vector length $\ell$, corruption threshold
      $t$, and number of signers $n$, the honest signers do the following.

      \begin{description}[style=nextline]

        \item[{$\vec{\alpha} = (1, \alpha)^{\mathsf{T}} \in \Zp^2$ and
          $\vec{\beta} = (1, \beta)^{\mathsf{T}} \in \Zp^2$}] Issue two calls
          to $\algo{RandShare}$ to compute $t$-additive secret shares
          $(\shr{\alpha}, \shr{\beta})$.

        \item[{$\mat{K} \in \Zp^{(\ell+1)\times 2}$}] Issue
          $2(\ell+1)$ calls to $\algo{RandShare}$ to get $t$-additive secret
          shares $\shr{\mat{K}} = (\shr{\kappa_{1,1}}, \ldots,
          \shr{\kappa_{\ell+1,2}})$.

        \item[{$\mat{\Gamma}, \mat{\Delta} \in \Zp^{2\times 2}$}]
          Invoke $\algo{RandShare}$ to obtain $t$-additive secret shares
          $(\shr{\gamma_{1,1}},\ldots, \shr{\gamma_{2,2}},
          \shr{\delta_{1,1}},\ldots, \shr{\delta_{2,2}})$. If no $\bot$ occurs
          during randomness generation, continue.

        \item[{$\vec{\mu} = \mat{K}\vec{\alpha}$}] Invoke
          $\algo{Mul}$ with shares $\shr{\alpha}$ and $\shr{\kappa_{i, 2}}$,
          for $i \in [\ell+1]$, to compute $t$-additive secret shares
          $(\shr{\kappa_{1, 2}\alpha},\ldots, \shr{\kappa_{\ell+1,2}\alpha})$.
          If no $\bot$ occurs, call each signer locally adds its shares
          $\shr{\kappa_{i, 1}}$ and $\shr{\kappa_{i, 2}\alpha}$ to obtain
          $t$-additive secret shares $(\shr{\mu_1},\ldots, \shr{\mu_{\ell+1}})$
          with $\mu_i = \kappa_{i, 1}+ \kappa_{i, 2}\alpha$ for all $i \in
          [\ell+1]$.

        \item[{$\vec{\nu} = \mat{\Gamma}\vec{\alpha}$}] Invoke
          $\algo{Mul}$ with shares $\shr{\alpha}$ and $\shr{\gamma_{1, 2}}$,
          and with $\shr{\alpha}$ and $\shr{\gamma_{2, 2}}$, to compute
          $t$-additive secret shares $(\shr{\gamma_{1,
          2}\alpha},\shr{\gamma_{2,2}\alpha})$. If no $\bot$ occurs, each
          signer locally adds its shares $\shr{\gamma_{1, 1}}$ and
          $\shr{\gamma_{1, 2}\alpha}$, and with $\shr{\gamma_{2, 1}}$ and
          $\shr{\gamma_{2, 2}\alpha}$, to obtain $t$-additive secret shares
          $(\shr{\nu_1}, \shr{\nu_2})$ with $\nu_1 = \gamma_{1, 1}+ \gamma_{1,
          2}\alpha$ and $\nu_2 = \gamma_{2, 1}+ \gamma_{2, 2}\alpha$.

        \item[{$\vec{\omega} = \mat{\Delta}\vec{\alpha}$}] Follow the
          same steps as for $\vec{\nu}$, with $\mat{\Gamma}$ replaced by
          $\mat{\Delta}$ and $\vec{\nu}$ replaced by $\vec{\omega}$.

        \item[{$\vec{\xi} = \vec{\beta}^{\mathsf{T}}\mat{\Gamma}$}]
          Invoke $\algo{Mul}$ with shares $\shr{\beta}$ and $\shr{\gamma_{2,
          1}}$, and with $\shr{\beta}$ and $\shr{\gamma_{2, 2}}$, to get
          $t$-additive secret shares $(\shr{\beta\gamma_{2,
          1}},\shr{\beta\gamma_{2,2}})$. If no $\bot$ occurs, each signer
          locally adds its shares $\shr{\gamma_{1, 1}}$ and
          $\shr{\beta\gamma_{2, 1}}$, and with $\shr{\gamma_{1, 2}}$ and
          $\shr{\beta\gamma_{2, 2}}$, to obtain $t$-additive secret shares
          $(\shr{\xi_1}, \shr{\xi_2})$ with $\xi_1 = \gamma_{1, 1}+
          \beta\gamma_{2, 1}$ and $\xi_2 = \gamma_{1, 2}+ \beta\gamma_{2, 2}$.

        \item[{$\vec{\upsilon} = \vec{\beta}^{\mathsf{T}}\mat{\Delta}$}]
          Follow the same steps as for $\vec{\xi}$, with $\mat{\Gamma}$
          replaced by $\mat{\Delta}$ and $\vec{\xi}$ replaced by
          $\vec{\upsilon}$.

        \item[{$(\kappa_{1, 1}\gen, \kappa_{1,2}\gen)$, $\vec{b} = (\gen,
          \beta\gen)$, $\vec{x} = (x_1, x_2) = (\xi_1\gen, \xi_2\gen)$,
        $\vec{y} = (y_1, y_2) = (\upsilon_1\gen, \upsilon_2\gen)$}] Each
        signer locally multiplies by the generator $\gen$ and shares
        $\shr{\kappa_{1, 1}}$, $\shr{\kappa_{1, 2}}$, $\shr{\beta}$,
        $\shr{\xi_1}$, $\shr{\xi_2}$, $\shr{\upsilon_1}$, and
        $\shr{\upsilon_2}$, to compute the corresponding $t$-additive shares.

      \item[{$\vec{\widetilde{a}} = (\ggen, \alpha\ggen)$, $\vec{\widetilde{u}}
        = (\mu_1\ggen, \ldots, \mu_{\ell+1}\ggen)$, $\vec{\widetilde{v}} =
      (\nu_1\ggen, \nu_2\ggen)$, $\vec{\widetilde{w}} = (\omega_1\ggen,
    \omega_2\ggen)$}] Each signer locally multiplies by the generator $\ggen$
    its shares $\shr{\alpha}$, $\shr{\mu_1} \ldots \shr{\mu_{\ell+1}}$,
    $\shr{\nu_1}$, $\shr{\nu_2}$, $\shr{\omega_1}$, and $\shr{\omega_2}$, to
    compute the corresponding $t$-additive shares.

  \item[{$\pk = (\vec{\widetilde{a}}, \vec{\widetilde{u}}, \vec{\widetilde{v}},
    \vec{\widetilde{w}})$}] Call $\algo{Open}$ to compute and publish the
    public key.  If no $\bot$ occurs, proceed to the next step.

  \item[{$\shr{\sk} = (\shr{\mat{K}}, \shr{b}, \shr{\vec{x}},
    \shr{\vec{y}})$}] Conclude key generation by storing the shares, together
    with the pair $\langle \shr{\kappa_{1, 1}\gen},\shr{\kappa_{1, 2}\gen}
    \rangle$.

\end{description}

\item[$\shr{\SPSsig}_{\algo{OffSign}} \sample \algo{OffSign}(\shr{\sk})$ :]
  On input the secret key shares, the honest signers do the following:

  \begin{enumerate}

    \item Invoke $\algo{RandShare}$ to get $t$-additive shares $\shr{\rho}$ and
      $\shr{\sigma}$.

    \item If no $\bot$ occurs, call $\algo{Mul}$ with shares $\shr{\rho}$ and
      $\shr{\sigma}$ to compute $t$-additive shares $\shr{\rho\sigma}$.

    \item If this succeeds, invoke $\algo{ExpGrp}_1$ to compute $t$-additive
      shares $\shr{\rho{b}}$, $\shr{\rho\sigma{b}}$, $\shr{\rho\vec{x}}$, and
      $\shr{\rho\sigma\vec{y}}$.

    \item If $\algo{ExpGrp}_1$ does not abort, each signer locally computes
      $t$-additive shares $\shr{\rho\gen}$, $\shr{\rho\sigma\gen}$, and
      $\shr{\sigma\ggen}$ against the relevant generators.

    \item Each signer locally adds its shares to receive $t$-additive shares
      $\shr{\vec{z}^{\star}} = \shr{\rho\vec{x} + \rho\sigma\vec{y}}=
      \shr{\rho\vec{x}} +\shr{\rho\sigma\vec{y}}$.

    \item Conclude the offline phase by storing $(\shr{\sigma\ggen},
      \shr{\rho\gen}, \shr{\rho\sigma\gen}, \shr{\rho{b}}, \shr{\rho\sigma{b}},
      \shr{\vec{z}^{\star}})$.

  \end{enumerate}

\item[$\shr{\SPSsig}_{\algo{OnSign}} \sample \algo{OnSign}(\vec{\msg})$ :] On
  input message vector $\vec{\msg}$, the honest signers do the following:

  \begin{enumerate}

    \item For each $\msg_i \neq 0$, each signer locally multiplies its shares
      $\shr{\kappa_{i, 1}}$ and $\shr{\kappa_{i, 2}}$ to obtain $t$-additive
      shares $\shr{\kappa_{i, 1} \msg_i} = \shr{\kappa_{i, 1}} \msg_i$ and
      $\shr{\kappa_{i+1, 2} \msg_i} = \shr{\kappa_{i+1, 2}}\msg_i$.

    \item Each signer locally adds these to compute the $t$-additive shares
      $\shr{(\gen, \vec{\msg})\mat{K}}$, comprising:
          %
      \begin{equation*}
          %
        \left(\shr{\kappa_{1, 1}\gen} + \sum_{i=1, \msg_i \neq 0}^{\ell+1}
        \shr{\kappa_{i+1, 1}\msg_i} , \shr{\kappa_{1, 2}\gen} + \sum_{i=1,
      \msg_i \neq 0}^{\ell+1} \shr{\kappa_{i+1, 2}\msg_i}\right).
          %
    \end{equation*}

  \item Each signer performs one final local addition to compute the shares
    $\shr{\vec{z}} = \shr{(\gen, \vec{\msg})\mat{K} +
    \vec{z}^{\star}} = \shr{(\gen, \vec{\msg})\mat{K}}+
    \shr{\vec{z}^{\star}}$.

\end{enumerate}

\item[$\SPSsig \leftarrow \TSPScombine(\shr{\SPSsig})$ :] On input the
  signature shares, the honest signers do the following:

  \begin{enumerate}

    \item Query $\algo{Open}$ to compute group elements $\vec{r} = (\rho\gen,
      \rho{b})$ and $\vec{s} = (\rho\sigma\gen, \rho\sigma{b})$, and
      $\algo{Open}$ to compute $\widetilde{q} = \sigma\ggen$.

    \item Invoke $\algo{Open}$ to compute $\vec{z}= (\gen,
      \vec{\msg})\mat{K} + \vec{z^{\star}}$.

    \item If no $\bot$ occurs, return the valid signature $\SPSsig =
      (\widetilde{q}, \vec{r}, \vec{s}, \vec{z})$ on vector $\vec{\msg} =
      (\msg_1, \ldots, \msg_\ell)$.

  \end{enumerate}

\item[$\{0,1\} \leftarrow \SPSverify(\pk, \vec{\msg}, \SPSsig)$ :] On input
  signature $\SPSsig = (\widetilde{q}, \vec{r}, \vec{s}, \vec{z})$,
  vector $(\msg_1, \ldots,\msg_\ell) \in \group{1}^\ell$, and public key $\pk
  =(\vec{\widetilde{a}}, \vec{\widetilde{u}},
  \vec{\widetilde{v}},\vec{\widetilde{w}})$, do the following:

  \begin{enumerate}

    \item Check whether~\eqref{eq:kiltz-ver} holds, returning 1 if so and 0
      otherwise.

  \end{enumerate}

\end{description}

\end{protocolbox}

Both schemes therefore fall within the class covered by
\Cref{thm:tsps-construction-security}, giving the following.

\begin{theorem}\label{thm:tsps-security} Let $\algo{MPC}$ satisfy the
  properties of \Cref{sec:tsps-mpc}. Then the threshold structure preserving
  signature scheme of \Cref{sec:tsps-abe} is adaptive TS-UF-1 secure under the
  wUF-CMA security of the structure preserving signature scheme of
  \Cref{sec:tsps-sps-ggm}; and the threshold structure preserving signature
  scheme of \Cref{sec:tsps-kiltz} is adaptive TS-UF-1 secure under the wUF-CMA
  security of the structure preserving signature scheme of
\Cref{sec:tsps-sps-kiltz}.  \end{theorem}
