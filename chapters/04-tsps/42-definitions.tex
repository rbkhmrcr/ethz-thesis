\section{Definitions}\label{sec:tsps-definitions}

A threshold structure-preserving signature scheme distributes the signing key of
an SPS scheme $\SPS$ between $n$ signers, so that any $t+1$ of them can jointly produce a signature, and no
$t$ of them learn anything about the key.

The signers are the only participants in the protocol. They communicate over
authenticated point-to-point channels, together with a broadcast channel that
delivers the same message to every signer. There is no trusted dealer to
generate the key and hand out its shares, so the signing key is never held in
full by any party. A verifier holds only the public key, and needs to know neither $n$, nor
$t$, nor which signers took part. Signers may be arbitrarily malicious,
deviating from the protocol at will, and may collude to forge signatures or to
recover the signing key. We consider adaptive corruption: the adversary chooses
which signers to corrupt during protocol execution, subject to $|\Cor| \leq t$.
Under static corruption, it chooses them all before key generation. We write
$\Cor$ for the set of corrupted signers and $\mathcal{H}$ for its complement.

\subsection{Threshold structure-preserving signature schemes}

\begin{outline}
  \item notation: $\shr{x}$ for a sharing of $x$ between the $n$ signers, $x_i$
for signer $i$'s share (made precise in \Cref{sec:tsps-mpc})
  \item signing split into an offline phase (before the message is known) and an
online phase
  \item lead into the list of algorithms of $\TSPS$
\end{outline}

\begin{description}

  \item[{$(\pk, \shr{\sk}) \sample \TSPSgen(\secparam, \ell, t, n)$ :}] Run
jointly by the signers. On input the message length $\ell$, the corruption bound
$t$, and the number of signers $n > t$, outputs a public key $\pk$ and gives
each signer $i$ a share $\sk_i$ of the secret key.

  \item[{$\shr{\state} \sample \TSPSoffsign(\shr{\sk})$ :}] Run jointly by the
signers before the message is known. Gives each signer $i$ a share $\state_i$ of
the state for one future signature.

  \item[{$\SPSsig_i \leftarrow \TSPSonsign(\sk_i, \state_i, \vec{\msg})$ :}] Run
locally by signer $i$ on a message $\vec{\msg} \in \group{1}^\ell$. Outputs a
signature share $\SPSsig_i$.

  \item[{$\SPSsig \leftarrow \TSPScombine(\{\SPSsig_i\}_{i \in Q})$ :}] On input
signature shares from a set $Q$ of $t+1$ signers, outputs a signature
$\SPSsig$, without interaction.

  \item[$\{0,1\} \leftarrow \TSPSverify(\pk, \vec{\msg}, \SPSsig)$ :] Equal to
$\SPSverify$.

\end{description}

\paragraph{Correctness.} A threshold SPS scheme is correct if, for all $(\ell, t, n > t)$, all $(\pk, \shr{\sk}) \sample
\TSPSgen(\secparam, \ell, t, n)$, all $\shr{\state} \sample
\TSPSoffsign(\shr{\sk})$, all $\vec{\msg} \in \group{1}^\ell$, and every $Q
\subseteq [1,n]$ with $|Q| = t+1$, the following holds:
%
\begin{equation*}
%
  \SPSverify\left(\pk, \vec{\msg}, \TSPScombine\left(\left\{
\TSPSonsign(\sk_i, \state_i, \vec{\msg}) \right\}_{i \in Q}\right)\right) = 1.
%
\end{equation*}

\paragraph{Unforgeability.}
\begin{outline}
  \item threshold schemes have a hierarchy of unforgeability notions (cite
C:BCKMTZ22); we use adaptive TS-UF-1
  \item winning condition: valid signature on $\vec{\msg}^*$ with shares from
fewer than $t + 1 - |\Cor|$ honest signers, so these plus the corrupted key
shares fall short of the $t+1$ needed
  \item TS-UF-0 is weaker: counts a forgery only if no honest signer signed
$\vec{\msg}^*$; TS-UF-1 also counts forgeries on messages too few honest signers
signed, and is our target
  \item achievable because the online phase is local: issuing a share reveals
nothing further; and it is what a certifier needs (no signature unless
enough honest signers agreed to that message)
  \item three oracles: offline $\mathcal{O}_{\algo{Off}}$ runs $\TSPSoffsign$
with $\adv$ playing the corrupted signers and returns their view (without it the
adversary never sees any interaction, since all of it is offline); online
$\mathcal{O}_{\algo{On}}$ returns an honest signer's share; corruption
$\mathcal{O}_{\algo{Cor}}$
  \item each session $\mathsf{sid}$ used for one message only, since reusing a
state reuses the signing randomness
\end{outline}

The advantage of a PPT adversary $\adv$ is defined as follows:
%
\begin{equation*}
%
  \Adv_{\TSPS, \adv}^{\text{adp-TS-UF-1}}(\secparam) = \Pr[\adv\text{ wins}],
%
\end{equation*}
%
in Experiment~\ref{exp:tsps-adp-ts-uf-1}. A threshold SPS scheme is adaptive TS-UF-1 secure if this advantage is negligible for
every PPT adversary $\adv$.

\begin{experimentbox}{Threshold structure-preserving signature schemes: adp-TS-UF-1}{tsps-adp-ts-uf-1}

  \begin{enumerate}

    \item The adversary outputs $(n, t, \Cor)$ with $|\Cor| \leq t$, and the
challenger sets $\mathcal{H} := [1,n] \setminus \Cor$.

    \item The challenger generates $(\pk, \shr{\sk}) \sample \TSPSgen(\secparam,
\ell, t, n)$, and sends $\pk$, the shares $\{\pk_i\}_{i \in [1,n]}$ that the
signers broadcast to open it, and $\{\sk_i\}_{i \in \Cor}$ to $\adv$.

    \item The adversary is given oracle access to the following:

      \begin{description}

	\item[$\mathcal{O}_{\algo{Off}}(\mathsf{sid})$:] If $\mathsf{sid}$ has
been used before, return $\perp$. Otherwise run $\TSPSoffsign$ between all $n$
signers, with $\adv$ acting for $\Cor$, store the shares
$\shr{\state}_{\mathsf{sid}}$, and return the view of the corrupted signers.

	\item[$\mathcal{O}_{\algo{On}}(i, \mathsf{sid}, \vec{\msg})$:] If $i
\notin \mathcal{H}$, $\mathsf{sid}$ was not initialised by
$\mathcal{O}_{\algo{Off}}$, or $\mathsf{sid}$ has been used with a message other
than $\vec{\msg}$, return $\perp$. Otherwise compute $\SPSsig_i \leftarrow
\TSPSonsign(\sk_i, \state_{\mathsf{sid},i}, \vec{\msg})$, add $i$ to
$S(\vec{\msg})$, and return $\SPSsig_i$.

	\item[$\mathcal{O}_{\algo{Cor}}(k)$:] If $k \in \Cor$ or $|\Cor| = t$,
return $\perp$. Otherwise move $k$ from $\mathcal{H}$ to $\Cor$, and return
$\sk_k$ together with every $\state_{\mathsf{sid},k}$.

      \end{description}

    \item The adversary outputs $(\vec{\msg}^*, \SPSsig^*)$.

  \end{enumerate}

  $\adv$ wins the experiment if $\SPSverify(\pk, \vec{\msg}^*, \SPSsig^*) = 1$
and $|S(\vec{\msg}^*) \cap \mathcal{H}| < t + 1 - |\Cor|$, where
$S(\vec{\msg})$ records which signers have issued a signature share on
$\vec{\msg}$.

\end{experimentbox}

\begin{outline}
  \item unforgeability is the only property we prove
  \item key secrecy follows from MPC privacy: signers only open $\pk$ and
signatures, so at most $t$ signers learn nothing about $\sk$ beyond those
  \item compatibility: $\TSPSverify = \SPSverify$ and a combined signature is an
ordinary signature under uniform randomness, so a verifier can't tell it was
produced by a threshold of signers
\end{outline}

\subsection{Secure multiparty computation}\label{sec:tsps-mpc}

\begin{outline}
  \item schemes are sequences of calls to an MPC protocol run by the $n$
signers; we only use its interfaces and stated properties
  \item sharing $\shr{\alpha}$ of $\alpha \in \Zp$: linear $(t+1, n)$ threshold
sharing; signer $i$ holds $\alpha_i$, any $t+1$ shares reconstruct by a fixed
linear combination, any $t$ are independent of $\alpha$
  \item linearity is the only property we rely on: makes additions and
exponentiations of public group elements by shared exponents local, which is what
makes the online phase local; any linear scheme works, so each protocol can use
the sharing that suits its corruption setting
  \item $\group{}$ for either source group; sharing $\shr{a}$ of $a \in
\group{}$ defined the same way, shares in $\group{}$, reconstruction the
product; shared as a single group element so group operations on shares stay
local
  \item sharings include authentication information (e.g.\ MAC shares) for
malicious security, so a modified share is detected at opening
  \item in protocol boxes $\shr{x}$ also stands for the signer's own share
\end{outline}

\begin{outline}
  \item honest majority ($t < n/2$): Shamir, $\alpha_i = f(i)$ for random $f$ of
degree $t$ with $f(0) = \alpha$, reconstruct by interpolation
  \item dishonest majority (e.g.\ MASCOT): additive, $\alpha_i$ uniform with
$\sum_i \alpha_i = \alpha$; this is $t = n-1$, so every signer takes part in
signing
  \item constructions stated for general $(t+1, n)$
\end{outline}

The protocol provides the following interfaces, each run by all signers
together. The interfaces over $\group{}$ exist once for each source group, and
we write $\algo{ExpGrp}_1$ for the one over $\group{1}$.

  \begin{description}

    \item[$\shr{\alpha} \sample \algo{RandShare}()$ :] Outputs a sharing of a
uniform $\alpha \sample \Zp$ that no signer learns.

    \item[$\shr{\gamma} \sample \algo{Mul}(\shr{\alpha}, \shr{\beta})$ :] Outputs
a sharing of $\gamma = \alpha\beta$.

    \item[$\shr{\gamma} \sample \algo{Inv}(\shr{\alpha})$ :] On input a sharing
of $\alpha \neq 0$, outputs a sharing of $\gamma = \alpha^{-1}$.

    \item[$\shr{c} \sample \algo{ExpGrp}(\shr{\alpha}, \shr{b})$ :] On input a
sharing of $\alpha \in \Zp$ and a sharing of $b \in \group{}$, outputs a sharing
of $c = b^\alpha$.

    \item[$\alpha \leftarrow \algo{Open}(\shr{\alpha})$ :] Reconstructs $\alpha$
and gives it to every signer, or to a designated party. Applies equally to a
sharing of a group element.

  \end{description}


Linear operations are not interfaces, since they need no interaction. Given
sharings $\shr{\alpha}$, $\shr{\beta}$ and public $\rho, \sigma, \tau \in \Zp$,
each signer applies the combination to its own shares to obtain $\shr{\rho\alpha
+ \sigma\beta + \tau} = \rho\shr{\alpha} + \sigma\shr{\beta} + \tau$. In a group,
given sharings $\shr{a}$, $\shr{b}$ and a public group element $d$, each signer
obtains $\shr{a}^{\rho}\shr{b}^{\sigma}d$. In particular, raising a public group
element to a shared exponent, $d^{\shr{\alpha}}$, gives a sharing of
$d^{\alpha}$~\cite{FOCS:Feldman87,LC:ADEO21}. This is how the signers obtain
shares of group elements such as $\gen^{\alpha}$ from shares of $\alpha$, and it
is what makes our online phase local, since the messages are public group
elements. Under malicious security, the same operations are applied to the
authentication information.

\begin{outline}
  \item $\algo{ExpGrp}$ is the only interface that mixes the two domains; only
needed when the secret key contains group elements ($\KPW$); for a key of field
elements ($\AGHO$) $\algo{Mul}$ is enough; it lets such keys be used as they are,
without replacing them by their discrete logarithms
  \item realised as Smart and Talibi Alaoui do, Beaver-style (cite IMA:SmaTal19);
explain Beaver multiplication: triples $(a, b, ab)$ made in preprocessing, each
multiplication uses one triple and two openings
  \item $\algo{ExpGrp}$ consumes a mixed triple $(\shr{\eta}, \shr{h},
\shr{h^\eta})$, $\eta \sample \Zp$, $h \sample \group{}$; signers open $\epsilon
= \alpha - \eta$ and $\delta = b h^{-1}$ (uniform, reveal nothing) and compute
locally
      %
      \begin{equation*}
      %
	\shr{b^{\alpha}} = \shr{h^{\eta}} \cdot \shr{h}^{\epsilon} \cdot
\delta^{\shr{\eta}} \cdot \delta^{\epsilon}
      %
      \end{equation*}
      %
      with the public term $\delta^{\epsilon}$ added by one designated signer
  \item triple obtained from an ordinary $\Zp$ triple by raising a generator to
the shares of its second and third entries; this, plus group element sharings
and their authentication information, is what we added to MP-SPDZ
\end{outline}

The remaining interface, $\algo{Inv}$, is built from $\algo{Mul}$ and
$\algo{Open}$: the signers sample $\shr{\beta} \sample \algo{RandShare}()$, open
$\gamma = \alpha\beta$, which is uniform and so reveals nothing, and take
$\shr{\alpha^{-1}} = \gamma^{-1}\shr{\beta}$. Where a scheme samples from
$\Zp^*$, the signers sample from $\Zp$ and abort if an opened $\gamma$ is zero,
which happens with probability $1/p$. Where no inverse is taken, sampling from
$\Zp$ instead of $\Zp^*$ changes the distribution only negligibly.

Finally, we require the MPC protocol to be private and secure with
abort against up to $t$ adaptively corrupted signers. Privacy means that the
view of the corrupted signers, over any sequence of calls, can be simulated by a
PPT simulator $\algo{Sim}$ from their own inputs and outputs and the values
opened. Security with abort means that corrupted signers may make a call return
$\bot$ to every honest signer, but cannot make it return a wrong output. Neither
protocol we benchmark is proven secure against adaptive corruption, so for them
our results hold under static corruption, for which our proof has no loss.
